# TokenHUB 本地大模型（P1+P2）Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 让 TokenHUB 能识别本机 NVIDIA/AMD 显卡、按需安装 llama.cpp 运行时、加载本地 GGUF 模型并作为可调优先级的供应商接入路由，同时提供 HuggingFace 浏览与断点续传下载。

**Architecture:** 新增不依赖 Electron 的 `core/local/` 子系统（硬件、运行时安装、模型库、下载、进程管理），本地实例以 `llama-server` 子进程形式只绑 `127.0.0.1`，通过新增的 `local` 预设作为 OpenAI 兼容供应商接入现有 Hub；主进程用独立 IPC 频道推送本地快照，渲染进程新增「本地模型」页面。

**Tech Stack:** TypeScript、Node 24（内置 fetch/ReadableStream）、Electron 主进程、Hono 网关、React 19 + Tailwind 4、Vitest、Playwright、PowerShell（Windows 硬件查询与 zip 解压）。

**Spec:** `docs/superpowers/specs/2026-11-22-local-llm-design.md`

## Global Constraints

- 所有本地服务（Hub、llama-server）只绑 `127.0.0.1`；本地推理端口默认 8788，可配置。
- 密钥（供应商 Key、本地 Key、HF Token）只存 `secrets.bin`；禁止明文落盘、写日志或放进 IPC 返回值。
- 本地供应商无 Key：转发时 `apiKey` 为空字符串，`forward.ts` 不得生成 `Authorization` / `x-api-key` 头；云端供应商缺 Key 仍必须 401。
- 只有「已加载」的本地模型进入 `models.cached` 与 catalog；停止、崩溃、卸载后立即清空。
- GPU 显存必须优先读注册表 `HardwareInformation.qwMemorySize`；`AdapterRAM` 仅在读不到时兜底并标记估算。
- 运行时与模型都落在用户数据目录，不进安装包；`build.files` 不改。
- 新增依赖从简：不引入原生模块、图表库或 zip 库；网络用注入的 `fetch`，解压用 PowerShell `Expand-Archive`。
- 项目约定：未得到用户明确要求，不执行 `git add` / `git commit` / `git push`。每个任务末尾的「检查点」只做 `git status` 与测试复核，提交需用户确认。
- 验证门（每个任务至少跑相关单测；最终全跑）：`npm run typecheck`、`npm test`、`npm run build`、`npx playwright test`。
- P3 范围（不在本计划）：ROCm 后端开放、mmproj 视觉、空闲自动卸载、VRAM 估算、GGUF 元数据预解析、`-ngl` 界面调节。

```ts
import type { BackendPreference, HfEndpointKind, LocalSettings } from './types'

export const DEFAULT_LOCAL_SETTINGS: LocalSettings = {
  modelsDir: '',
  localPort: 8788,
  backend: 'auto',
  hfEndpoint: 'official',
  hfEndpointCustom: '',
  githubMirror: '',
  defaultContext: 8192
}

const BACKENDS = new Set<BackendPreference>(['auto', 'cuda', 'vulkan', 'rocm', 'cpu'])
const ENDPOINTS = new Set<HfEndpointKind>(['official', 'mirror', 'custom'])

function intInRange(raw: unknown, fallback: number, min: number, max: number): number {
  const value = typeof raw === 'number' ? raw : Number(raw)
  return Number.isInteger(value) && value >= min && value <= max ? value : fallback
}

export function normalizeLocalSettings(raw: unknown): LocalSettings {
  const input = (raw ?? {}) as Partial<LocalSettings>
  const backend = BACKENDS.has(input.backend as BackendPreference) ? (input.backend as BackendPreference) : 'auto'
  const hfEndpoint = ENDPOINTS.has(input.hfEndpoint as HfEndpointKind)
    ? (input.hfEndpoint as HfEndpointKind)
    : 'official'
  return {
    modelsDir: typeof input.modelsDir === 'string' ? input.modelsDir.trim() : '',
    localPort: intInRange(input.localPort, DEFAULT_LOCAL_SETTINGS.localPort, 1, 65535),
    backend,
    hfEndpoint,
    hfEndpointCustom:
      typeof input.hfEndpointCustom === 'string' ? input.hfEndpointCustom.trim().replace(/\/+$/, '') : '',
    githubMirror: typeof input.githubMirror === 'string' ? input.githubMirror.trim() : '',
    defaultContext: intInRange(input.defaultContext, DEFAULT_LOCAL_SETTINGS.defaultContext, 512, 262144)
  }
}

export function hfBaseUrl(settings: Pick<LocalSettings, 'hfEndpoint' | 'hfEndpointCustom'>): string {
  if (settings.hfEndpoint === 'mirror') return 'https://hf-mirror.com'
  if (settings.hfEndpoint === 'custom' && settings.hfEndpointCustom) return settings.hfEndpointCustom
  return 'https://huggingface.co'
}
```

```ts
import { createWriteStream } from 'node:fs'
import { mkdir, rename, stat } from 'node:fs/promises'
import { dirname } from 'node:path'
import { Readable } from 'node:stream'
import { pipeline } from 'node:stream/promises'

export type FetchFn = (url: string, init?: RequestInit) => Promise<Response>

export interface DownloadFileOptions {
  url: string
  dest: string
  headers?: Record<string, string>
  fetchFn: FetchFn
  signal?: AbortSignal
  onProgress?: (received: number, total: number | null) => void
}

function totalFromHeaders(headers: Headers, startAt: number): number | null {
  const range = headers.get('content-range')
  if (range) {
    const match = /\/(\d+)$/.exec(range)
    if (match) return Number(match[1])
  }
  const length = headers.get('content-length')
  if (length) return startAt + Number(length)
  return null
}

async function renameWithRetry(source: string, target: string): Promise<void> {
  for (let attempt = 0; ; attempt += 1) {
    try {
      await rename(source, target)
      return
    } catch (error) {
      if (attempt >= 4) throw error
      await new Promise((resolve) => setTimeout(resolve, 50 * (attempt + 1)))
    }
  }
}

export async function downloadFile(options: DownloadFileOptions): Promise<{ received: number; total: number | null }> {
  const part = `${options.dest}.part`
  const existing = await stat(part).then((info) => info.size).catch(() => 0)
  const headers: Record<string, string> = { ...(options.headers ?? {}) }
  if (existing > 0) headers.Range = `bytes=${existing}-`

  const response = await options.fetchFn(options.url, { headers, signal: options.signal })
  if (!response.ok) throw new Error(`download failed: HTTP ${response.status}`)

  const resumed = response.status === 206 && existing > 0
  const startAt = resumed ? existing : 0
  const total = totalFromHeaders(response.headers, startAt)

  await mkdir(dirname(options.dest), { recursive: true })
  const out = createWriteStream(part, { flags: resumed ? 'a' : 'w' })
  let received = startAt
  const source = Readable.fromWeb(response.body as never)
  await pipeline(
    source,
    async function* (chunks) {
      for await (const chunk of chunks) {
        const buffer = Buffer.isBuffer(chunk) ? chunk : Buffer.from(chunk as Uint8Array)
        received += buffer.length
        options.onProgress?.(received, total)
        yield buffer
      }
    },
    out
  )
  await renameWithRetry(part, options.dest)
  return { received, total }
}
```

```ts
import { execFile } from 'node:child_process'
import type { BackendPreference, GpuInfo, HardwareReport, LocalBackend } from './types'

export type CommandRunner = (file: string, args: string[], timeoutMs?: number) => Promise<string>

export function runCommand(file: string, args: string[], timeoutMs = 15_000): Promise<string> {
  return new Promise((resolve, reject) => {
    execFile(file, args, { windowsHide: true, timeout: timeoutMs, maxBuffer: 4 * 1024 * 1024 }, (error, stdout, stderr) => {
      if (error) reject(new Error(String(stderr || error.message).trim()))
      else resolve(stdout)
    })
  })
}

export const GPU_QUERY_SCRIPT = [

function numberOrNull(raw: unknown): number | null {

function vendorOf(name: string, pnpId: string): GpuInfo['vendor'] {
  const id = pnpId.toUpperCase()
  if (id.includes('VEN_10DE') || /nvidia/i.test(name)) return 'nvidia'
  if (id.includes('VEN_1002') || /amd|radeon/i.test(name)) return 'amd'
  if (id.includes('VEN_8086') || /intel/i.test(name)) return 'intel'
  return 'other'
}

const SOFTWARE_ADAPTER = /microsoft basic display|remote display|basic render|virtual display/i

export function parseGpuReport(rawJson: string): GpuInfo[] {
  const parsed = JSON.parse(rawJson) as { wmi?: RawWmi[] | RawWmi; reg?: RawReg[] | RawReg }
  const wmi = Array.isArray(parsed.wmi) ? parsed.wmi : parsed.wmi ? [parsed.wmi] : []
  const reg = Array.isArray(parsed.reg) ? parsed.reg : parsed.reg ? [parsed.reg] : []
  const gpus: GpuInfo[] = []
  const seen = new Set<string>()
  for (const item of wmi) {
    const name = typeof item.Name === 'string' ? item.Name.trim() : ''
    if (!name || SOFTWARE_ADAPTER.test(name) || seen.has(name)) continue
    seen.add(name)
    const pnp = typeof item.PNPDeviceID === 'string' ? item.PNPDeviceID : ''
    const match = reg.find((entry) => {
      const desc = typeof entry.Desc === 'string' ? entry.Desc.trim() : ''
      return desc === name || (desc && (name.includes(desc) || desc.includes(name)))
    })
    const registryVram = numberOrNull(match?.Vram)
    const adapterVram = numberOrNull(item.AdapterRAM)
    gpus.push({
      name,
      vendor: vendorOf(name, pnp),
      vramBytes: registryVram ?? adapterVram,
      vramEstimated: registryVram === null && adapterVram !== null,
      recommended: false
    })
  }
  const discrete = gpus.filter((gpu) => gpu.vendor === 'nvidia' || gpu.vendor === 'amd')
  const pool = discrete.length > 0 ? discrete : gpus
  const best = pool.reduce<GpuInfo | null>(
    (current, gpu) => (!current || (gpu.vramBytes ?? 0) > (current.vramBytes ?? 0) ? gpu : current),
    null
  )
  if (best) best.recommended = true
  return gpus
}

export function pickRecommendedBackend(gpus: GpuInfo[], preference: BackendPreference): LocalBackend {
  if (preference !== 'auto') return preference
  const gpu = gpus.find((item) => item.recommended) ?? gpus.find((item) => item.vendor !== 'other')
  if (!gpu) return 'cpu'
  return gpu.vendor === 'nvidia' ? 'cuda' : 'vulkan'
}

export async function detectHardware(
  run: CommandRunner,
  preference: BackendPreference,
  now: () => Date = () => new Date()
): Promise<HardwareReport> {
  try {
    const raw = await run('powershell.exe', ['-NoProfile', '-NonInteractive', '-Command', GPU_QUERY_SCRIPT])
    const gpus = parseGpuReport(raw)
    return { gpus, recommendedBackend: pickRecommendedBackend(gpus, preference), detectedAt: now().toISOString() }
  } catch (error) {
    return {
      gpus: [],
      recommendedBackend: preference === 'auto' ? 'cpu' : preference,
      detectedAt: now().toISOString(),
      error: error instanceof Error ? error.message : String(error)
    }
  }
}
```

```ts
import { mkdir, readdir, readFile, rename, rm, stat, writeFile } from 'node:fs/promises'
import { join } from 'node:path'
import type { FetchFn } from './http'
import type { LocalBackend, RuntimeInstall } from './types'

export interface ReleaseAsset {
  name: string
  browser_download_url: string
}

export interface ReleaseInfo {
  tag_name: string
  assets: ReleaseAsset[]
}

export interface InstallerPorts {
  download(options: { url: string; dest: string; onProgress?: (received: number, total: number | null) => void }): Promise<void>
  extract(zipPath: string, destDir: string): Promise<void>
  validate(exePath: string): Promise<string[]>
}

export function assetNamesFor(backend: LocalBackend, tag: string): string[] {
  if (backend === 'cuda') return [`llama-${tag}-bin-win-cuda-12.4-x64.zip`, 'cudart-llama-bin-win-cuda-12.4-x64.zip']
  return [`llama-${tag}-bin-win-${backend}-x64.zip`]
}

export function selectRuntimeRelease(
  releases: ReleaseInfo[],
  backend: LocalBackend
): { tag: string; assets: Array<{ name: string; url: string }> } | null {
  for (const release of releases) {
    const required = assetNamesFor(backend, release.tag_name)
    const found = required.map((name) => release.assets.find((asset) => asset.name === name))
    if (found.every((asset): asset is ReleaseAsset => Boolean(asset))) {
      return {
        tag: release.tag_name,
        assets: found.map((asset) => ({ name: asset!.name, url: asset!.browser_download_url }))
      }
    }
  }
  return null
}

export function mirroredUrl(url: string, mirror: string): string {
  const prefix = mirror.trim().replace(/\/+$/, '')
  return prefix ? `${prefix}/${url}` : url
}

const RELEASES_URL = 'https://api.github.com/repos/ggml-org/llama.cpp/releases?per_page=10'

export interface RuntimeInstallerOptions {
  dataDir: string
  ports: InstallerPorts
  fetchFn: FetchFn
  mirror: () => string
  now?: () => Date
}

export class RuntimeInstaller {
  private readonly now: () => Date

  constructor(private readonly options: RuntimeInstallerOptions) {
    this.now = options.now ?? (() => new Date())
  }

  private get runtimeRoot(): string {
    return join(this.options.dataDir, 'local', 'runtime')
  }

  private get recordFile(): string {
    return join(this.options.dataDir, 'local', 'runtime.json')
  }

  async listInstalled(): Promise<RuntimeInstall[]> {
    const text = await readFile(this.recordFile, 'utf8').catch(() => null)
    if (!text) return []
    try {
      const parsed = JSON.parse(text) as unknown
      return Array.isArray(parsed) ? (parsed as RuntimeInstall[]) : []
    } catch {
      return []
    }
  }

  async getExe(backend: LocalBackend): Promise<RuntimeInstall | null> {
    const installs = await this.listInstalled()
    return installs.find((install) => install.backend === backend) ?? null
  }

  private async saveRecord(install: RuntimeInstall): Promise<void> {
    const installs = (await this.listInstalled()).filter((item) => item.backend !== install.backend)
    installs.push(install)
    await mkdir(join(this.options.dataDir, 'local'), { recursive: true })
    await writeFile(this.recordFile, JSON.stringify(installs, null, 2), 'utf8')
  }

  async install(backend: LocalBackend): Promise<RuntimeInstall> {
    const response = await this.options.fetchFn(RELEASES_URL, {
      headers: { accept: 'application/vnd.github+json', 'user-agent': 'TokenHUB' }
    })
    if (!response.ok) throw new Error(`GitHub release 查询失败：HTTP ${response.status}`)
    const releases = (await response.json()) as ReleaseInfo[]
    const selected = selectRuntimeRelease(releases, backend)
    if (!selected) throw new Error(`没有找到包含 ${backend} Windows 资产的 llama.cpp 版本`)

    const downloadDir = join(this.options.dataDir, 'local', 'downloads', 'runtime')
    await mkdir(downloadDir, { recursive: true })
    const zips: string[] = []
    for (const asset of selected.assets) {
      const dest = join(downloadDir, asset.name)
      await this.options.ports.download({ url: mirroredUrl(asset.url, this.options.mirror()), dest })
      zips.push(dest)
    }

    const targetDir = join(this.runtimeRoot, backend, selected.tag)
    const staging = `${targetDir}.tmp`
    await rm(staging, { recursive: true, force: true })
    await mkdir(staging, { recursive: true })
    for (const zip of zips) await this.options.ports.extract(zip, staging)

    const exePath = await findExecutable(staging)
    if (!exePath) throw new Error('压缩包中没有找到 llama-server.exe')
    const devices = await this.options.ports.validate(exePath)

    await rm(targetDir, { recursive: true, force: true })
    await mkdir(join(this.runtimeRoot, backend), { recursive: true })
    await rename(staging, targetDir)
    const finalExe = exePath.replace(staging, targetDir)

    const install: RuntimeInstall = {
      backend,
      tag: selected.tag,
      installedAt: this.now().toISOString(),
      devices,
      sourceUrl: selected.assets.map((asset) => asset.url).join(' '),
      exePath: finalExe
    }
    await this.saveRecord(install)
    return install
  }

  async importZip(zipPath: string, backend: LocalBackend): Promise<RuntimeInstall> {
    const tag = `manual-${this.now().toISOString().replace(/[:.]/g, '-')}`
    const targetDir = join(this.runtimeRoot, backend, tag)
    const staging = `${targetDir}.tmp`
    await rm(staging, { recursive: true, force: true })
    await mkdir(staging, { recursive: true })
    await this.options.ports.extract(zipPath, staging)
    const exePath = await findExecutable(staging)
    if (!exePath) throw new Error('压缩包中没有找到 llama-server.exe')
    const devices = await this.options.ports.validate(exePath)
    await mkdir(join(this.runtimeRoot, backend), { recursive: true })
    await rename(staging, targetDir)
    const install: RuntimeInstall = {
      backend,
      tag,
      installedAt: this.now().toISOString(),
      devices,
      sourceUrl: zipPath,
      exePath: exePath.replace(staging, targetDir)
    }
    await this.saveRecord(install)
    return install
  }

  async remove(backend: LocalBackend): Promise<void> {
    const installs = await this.listInstalled()
    const kept = installs.filter((install) => install.backend !== backend)
    await rm(join(this.runtimeRoot, backend), { recursive: true, force: true })
    await mkdir(join(this.options.dataDir, 'local'), { recursive: true })
    await writeFile(this.recordFile, JSON.stringify(kept, null, 2), 'utf8')
  }
}

async function findExecutable(dir: string): Promise<string | null> {
  const entries = await readdir(dir, { withFileTypes: true })
  for (const entry of entries) {
    const path = join(dir, entry.name)
    if (entry.isFile() && entry.name.toLowerCase() === 'llama-server.exe') return path
    if (entry.isDirectory()) {
      const nested = await findExecutable(path)
      if (nested) return nested
    }
  }
  return null
}
```

```ts
import { readdir, stat } from 'node:fs/promises'
import { join, posix } from 'node:path'
import type { InstalledModel } from './types'

const SPLIT_PATTERN = /^(.+)-(\d{5})-of-(\d{5})\.gguf$/i
const QUANT_PATTERN = /(?:^|[-_.])(IQ\d+(?:_[A-Z0-9]+)+|Q\d+(?:_[A-Z0-9]+)+|Q\d+|F16|F32|BF16)(?=[-_.]|$)/i

export function splitInfo(fileName: string): { base: string; part: number; total: number } | null {
  const match = SPLIT_PATTERN.exec(fileName)
  if (!match) return null
  return { base: match[1], part: Number(match[2]), total: Number(match[3]) }
}

export function quantFromFileName(fileName: string): string | null {
  const match = QUANT_PATTERN.exec(fileName)
  return match ? match[1].toUpperCase() : null
}

export function isSafeRelativePath(rel: string): boolean {
  const normalized = rel.replace(/\\/g, '/')
  if (!normalized || normalized.startsWith('/') || /^[a-z]:/i.test(normalized)) return false
  return normalized.split('/').every((part) => part !== '' && part !== '.' && part !== '..')
}

export function modelIdFromName(name: string): string {
  const stem = name.replace(/\.gguf$/i, '')
  const id = stem.replace(/[^\w.@:+-]+/g, '-').replace(/^-+|-+$/g, '')
  return id || 'local-model'
}

function splitParts(base: string, total: number): string[] {
  return Array.from({ length: total }, (_, index) => `${base}-${String(index + 1).padStart(5, '0')}-of-${String(total).padStart(5, '0')}.gguf`)
}

export function groupGgufFiles(inputs: Array<{ relative: string; size: number }>): InstalledModel[] {
  const splits = new Map<string, { base: string; total: number; dir: string; parts: Map<number, { relative: string; size: number }> }>()
  const singles: Array<{ relative: string; size: number }> = []
  for (const input of inputs) {
    const relative = input.relative.replace(/\\/g, '/')
    const name = posix.basename(relative)
    if (name.toLowerCase().startsWith('mmproj') || !isSafeRelativePath(relative)) continue
    const split = splitInfo(name)
    if (!split) {
      singles.push({ relative, size: input.size })
      continue
    }
    const dir = posix.dirname(relative)
    const key = `${dir}/${split.base}/${split.total}`
    const entry = splits.get(key) ?? { base: split.base, total: split.total, dir, parts: new Map() }
    entry.parts.set(split.part, { relative, size: input.size })
    splits.set(key, entry)
  }

  const models: InstalledModel[] = []
  for (const entry of splits.values()) {
    const complete = splitParts(entry.base, entry.total).every((_, index) => entry.parts.has(index + 1))
    if (!complete) continue
    const parts = [...entry.parts.values()]
    models.push(buildModel(entry.dir, entry.base, parts, entry.total))
  }
  for (const single of singles) {
    const dir = posix.dirname(single.relative)
    const base = posix.basename(single.relative).replace(/\.gguf$/i, '')
    models.push(buildModel(dir, base, [single], 1))
  }

  const used = new Map<string, number>()
  return models
    .map((model) => {
      const count = used.get(model.id) ?? 0
      used.set(model.id, count + 1)
      return count === 0 ? model : { ...model, id: `${model.id}@${repoDirOf(model.path) || 'root'}`.replace(/[^\w.@:+-]+/g, '-') }
    })
    .sort((a, b) => a.id.localeCompare(b.id))
}

function buildModel(dir: string, base: string, parts: Array<{ relative: string; size: number }>, partCount: number): InstalledModel {
  const repo = dir && dir !== '.' && dir.includes('/') ? dir.split('/').slice(0, 2).join('/') : dir && dir !== '.' ? dir : undefined
  return {
    id: modelIdFromName(base),
    path: parts[0].relative,
    sizeBytes: parts.reduce((sum, part) => sum + part.size, 0),
    parts: partCount,
    repo,
    quant: quantFromFileName(base) ?? undefined,
    loaded: false
  }
}

function repoDirOf(path: string): string {
  const dir = posix.dirname(path)
  return dir === '.' ? '' : dir.split('/').slice(0, 2).join('-')
}

async function walk(root: string, dir: string, depth: number, out: Array<{ relative: string; size: number }>): Promise<void> {
  if (depth > 3) return
  const entries = await readdir(dir, { withFileTypes: true }).catch(() => [])
  for (const entry of entries) {
    const full = join(dir, entry.name)
    if (entry.isDirectory()) {
      await walk(root, full, depth + 1, out)
      continue
    }
    if (!entry.isFile() || !/\.gguf$/i.test(entry.name) || /\.part$/i.test(entry.name)) continue
    const info = await stat(full).catch(() => null)
    if (!info) continue
    out.push({ relative: full.slice(root.length + 1), size: info.size })
  }
}
```

```ts
import type { FetchFn } from './http'
import type { LocalBackend, LocalRuntimeStatus, RuntimeInstall } from './types'

export interface ChildHandle {
  pid?: number
  kill(): boolean
  once(event: 'exit', cb: (code: number | null) => void): void
  stdout?: NodeJS.ReadableStream | null
  stderr?: NodeJS.ReadableStream | null
}

export interface RuntimeSpawnPorts {
  spawn(file: string, args: string[], options: { windowsHide: boolean }): ChildHandle
  killTree?(pid: number): Promise<void>
}

export interface LocalRuntimeOptions {
  getInstall: (backend: LocalBackend) => RuntimeInstall | null
  getPort: () => number
  spawnPorts: RuntimeSpawnPorts
  fetchFn: FetchFn
  healthPollMs?: number
  loadTimeoutMs?: number
  now?: () => Date
  onState?: (status: LocalRuntimeStatus) => void
}

const LOG_LINES = 200
const TAIL_LINES = 20

export function buildLlamaArgs(input: { modelPath: string; port: number; contextSize: number; parallel: number }): string[] {
  return [
    '-m', input.modelPath,
    '--host', '127.0.0.1',
    '--port', String(input.port),
    '-c', String(input.contextSize),
    '-ngl', '99',
    '-np', String(input.parallel),
    '--jinja'
  ]
}

function idleStatus(): LocalRuntimeStatus {
  return {
    phase: 'stopped',
    backend: null,
    tag: null,
    modelId: null,
    modelPath: null,
    port: null,
    contextLength: null,
    error: null,
    logTail: [],
    startedAt: null
  }
}

export class LocalRuntime {
  private status = idleStatus()
  private child: ChildHandle | null = null
  private logs: string[] = []
  private exitHook: (() => void) | null = null
  private stopping = false
  private readonly now: () => Date

  constructor(private readonly options: LocalRuntimeOptions) {
    this.now = options.now ?? (() => new Date())
  }

  getStatus(): LocalRuntimeStatus {
    return { ...this.status, logTail: [...this.status.logTail] }
  }

  setOnExit(cb: () => void): void {
    this.exitHook = cb
  }

  private emit(status: Partial<LocalRuntimeStatus>): void {
    this.status = { ...this.status, ...status }
    this.options.onState?.(this.getStatus())
  }

  private appendLog(line: string): void {
    const text = line.trimEnd()
    if (!text) return
    this.logs.push(text)
    if (this.logs.length > LOG_LINES) this.logs.shift()
    this.status = { ...this.status, logTail: this.logs.slice(-TAIL_LINES) }
    this.options.onState?.(this.getStatus())
  }

  async load(model: { modelId: string; modelPath: string }, backend: LocalBackend, contextSize: number): Promise<void> {
    await this.stop()
    const install = this.options.getInstall(backend)
    if (!install) {
      this.emit({ phase: 'error', error: `未安装 ${backend} 运行时`, modelId: model.modelId, backend })
      return
    }
    const port = this.options.getPort()
    this.logs = []
    this.stopping = false
    this.emit({
      phase: 'starting',
      backend,
      tag: install.tag,
      modelId: model.modelId,
      modelPath: model.modelPath,
      port,
      contextLength: contextSize,
      error: null,
      logTail: [],
      startedAt: this.now().toISOString()
    })

    const args = buildLlamaArgs({ modelPath: model.modelPath, port, contextSize, parallel: 1 })
    let child: ChildHandle
    try {
      child = this.options.spawnPorts.spawn(install.exePath, args, { windowsHide: true })
    } catch (error) {
      this.emit({ phase: 'error', error: error instanceof Error ? error.message : String(error) })
      return
    }
    this.child = child
    for (const stream of [child.stdout, child.stderr]) {
      stream?.on('data', (chunk: Buffer | string) => this.appendLog(String(chunk)))
    }
    let exited = false
    child.once('exit', (code) => {
      exited = true
      if (this.child === child) this.child = null
      if (!this.stopping) {
        this.emit({ phase: 'error', error: `llama-server 退出（code ${code ?? 'unknown'}）` })
        this.exitHook?.()
      }
    })

    const deadline = Date.now() + (this.options.loadTimeoutMs ?? 180_000)
    const pollMs = this.options.healthPollMs ?? 500
    const base = `http://127.0.0.1:${port}`
    while (Date.now() < deadline) {
      if (exited) return
      try {
        const response = await this.options.fetchFn(`${base}/health`)
        if (response.ok) {
          let actualContext = contextSize
          try {
            const props = (await (await this.options.fetchFn(`${base}/props`)).json()) as {
              default_generation_settings?: { n_ctx?: unknown }
            }
            const nCtx = props.default_generation_settings?.n_ctx
            if (typeof nCtx === 'number' && nCtx > 0) actualContext = Math.floor(nCtx)
          } catch {
            // /props 失败不影响已就绪的推理服务。
          }
          this.emit({ phase: 'running', error: null, contextLength: actualContext })
          return
        }
      } catch {
        // 进程还没开始监听，继续轮询。
      }
      await new Promise((resolve) => setTimeout(resolve, pollMs))
    }
    this.emit({ phase: 'error', error: '等待 llama-server 就绪超时（180s）' })
    await this.kill()
  }

  private async kill(): Promise<void> {
    const child = this.child
    this.child = null
    if (!child) return
    // Windows 上没有需要优雅收尾的会话：先 kill，再用 taskkill 兜底整棵进程树。
    child.kill()
    if (child.pid !== undefined) await this.options.spawnPorts.killTree?.(child.pid)
  }

  async stop(): Promise<void> {
    this.stopping = true
    await this.kill()
    this.logs = []
    this.emit({ ...idleStatus(), phase: this.status.phase === 'not-installed' ? 'not-installed' : 'stopped' })
    this.stopping = false
  }

  async shutdown(): Promise<void> {
    await this.stop()
  }
}
```

```ts
        this.settings = normalizeLocalSettings(JSON.parse(text))

        await this.options.logger?.warn('local.json 无法解析，已回退默认设置')

    await mkdir(this.modelsDir, { recursive: true }).catch(() => {})
    this.installs = await this.installer.listInstalled()
    await this.detect()
    await this.refreshModels()
    await this.options.hub.syncLocalProvider({ baseUrl: this.baseUrl, models: [] })
  }

  get baseUrl(): string {
    return `http://127.0.0.1:${this.settings.localPort}/v1`
  }

  private installFor(backend: LocalBackend): RuntimeInstall | null {
    return this.installs.find((install) => install.backend === backend) ?? null
  }

  private preferredInstall(): RuntimeInstall | null {
    const preferred = this.settings.backend === 'auto' ? this.hardware.recommendedBackend : this.settings.backend
    return this.installFor(preferred) ?? this.installs[0] ?? null
  }

  getSnapshot(): LocalSnapshot {
    return {
      settings: { ...this.settings },
      hardware: structuredClone(this.hardware),
      install: this.preferredInstall(),
      runtime: this.runtime.getStatus(),
      models: this.models.map((model) => ({ ...model })),
      downloads: [],
      hfTokenSet: this.options.secrets.hasHfToken()
    }
  }

  private snapshot(): LocalSnapshot {
    return this.getSnapshot()
  }

  private scheduleEmit(): void {
    if (this.emitTimer) return
    this.emitTimer = setTimeout(() => {
      this.emitTimer = null
      this.options.onSnapshot?.(this.snapshot())
    }, 300)
  }

  private emitNow(): void {
    if (this.emitTimer) {
      clearTimeout(this.emitTimer)
      this.emitTimer = null
    }
    this.options.onSnapshot?.(this.snapshot())
  }

  async detect(): Promise<LocalSnapshot> {
    const detect = this.options.detect ?? ((preference) => detectHardware(runCommand, preference, this.now))
    this.hardware = await detect(this.settings.backend)
    this.emitNow()
    return this.snapshot()
  }

  async detectHardware(): Promise<LocalSnapshot> {
    return this.detect()
  }

  async installRuntime(backend?: LocalBackend): Promise<LocalSnapshot> {
    const target =
      backend ?? (this.settings.backend === 'auto' ? this.hardware.recommendedBackend : this.settings.backend)
    const install = await this.installer.install(target)
    this.installs = (await this.installer.listInstalled()).filter((item) => item.backend !== target).concat(install)
    this.emitNow()
    return this.snapshot()
  }

  async removeRuntime(backend: LocalBackend): Promise<LocalSnapshot> {
    await this.runtime.stop()
    await this.installer.remove(backend)
    this.installs = this.installs.filter((install) => install.backend !== backend)
    await this.options.hub.syncLocalProvider({ baseUrl: this.baseUrl, models: [] })
    this.emitNow()
    return this.snapshot()
  }

  async importRuntimeZip(zipPath: string): Promise<LocalSnapshot> {
    const backend = this.hardware.recommendedBackend
    const install = await this.installer.importZip(zipPath, backend)
    this.installs = (await this.installer.listInstalled()).filter((item) => item.backend !== backend).concat(install)
    this.emitNow()
    return this.snapshot()
  }

  async refreshModels(): Promise<LocalSnapshot> {
    this.models = await scanModels(this.modelsDir)
    const loadedId = this.runtime.getStatus().modelId
    this.models = this.models.map((model) => ({ ...model, loaded: model.id === loadedId }))
    this.emitNow()
    return this.snapshot()
  }

  async loadModel(id: string, contextSize?: number): Promise<LocalSnapshot> {
    const model = this.models.find((candidate) => candidate.id === id)
    if (!model) throw new Error(`unknown local model: ${id}`)
    const install = this.preferredInstall()
    if (!install) throw new Error('请先安装 llama.cpp 运行时')
    await this.runtime.load({ modelId: model.id, modelPath: join(this.modelsDir, model.path) }, install.backend, contextSize ?? this.settings.defaultContext)
    const status = this.runtime.getStatus()
    if (status.phase !== 'running') {
      this.emitNow()
      return this.snapshot()
    }
    this.models = this.models.map((candidate) => ({ ...candidate, loaded: candidate.id === model.id }))
    await this.options.hub.syncLocalProvider({
      baseUrl: this.baseUrl,
      models: [model.id],
      meta: { [model.id]: { contextLength: status.contextLength, vision: false } }
    })
    this.emitNow()
    return this.snapshot()
  }

  async stopModel(): Promise<LocalSnapshot> {
    await this.runtime.stop()
    this.models = this.models.map((model) => ({ ...model, loaded: false }))
    await this.options.hub.syncLocalProvider({ baseUrl: this.baseUrl, models: [] })
    this.emitNow()
    return this.snapshot()
  }

  async deleteModel(id: string): Promise<LocalSnapshot> {
    const model = this.models.find((candidate) => candidate.id === id)
    if (!model) throw new Error(`unknown local model: ${id}`)
    if (this.runtime.getStatus().modelId === id) await this.stopModel()
    const { rm } = await import('node:fs/promises')
    await rm(join(this.modelsDir, model.path), { force: true })
    return await this.refreshModels()
  }

  private async onRuntimeExit(): Promise<void> {
    this.models = this.models.map((model) => ({ ...model, loaded: false }))
    await this.options.hub.syncLocalProvider({ baseUrl: this.baseUrl, models: [] }).catch(() => {})
    this.emitNow()
  }

  async updateSettings(patch: Partial<LocalSettings>): Promise<LocalSnapshot> {
    const next = normalizeLocalSettings({ ...this.settings, ...patch })
    const portChanged = next.localPort !== this.settings.localPort
    const dirChanged = next.modelsDir !== this.settings.modelsDir || Boolean(this.options.modelsDirOverride)
    this.settings = next
    await mkdir(this.modelsDir, { recursive: true }).catch(() => {})
    await writeFile(this.settingsFile, JSON.stringify(this.settings, null, 2), 'utf8')
    if (portChanged) await this.stopModel()
    if (dirChanged) await this.refreshModels()
    this.emitNow()
    return this.snapshot()
  }

  async setHfToken(token: string | null): Promise<LocalSnapshot> {
    await this.options.secrets.setHfToken(token)
    this.emitNow()
    return this.snapshot()
  }

  async shutdown(): Promise<void> {
    if (this.emitTimer) clearTimeout(this.emitTimer)
    await this.runtime.shutdown()
  }

  getModelsDir(): string {
    return this.modelsDir
  }

  getSettings(): LocalSettings {
    return { ...this.settings }
  }
}
```

```ts
  spawn: (file, args, options) =>
    spawn(file, args, { ...options, stdio: ['ignore', 'pipe', 'pipe'] }) as unknown as ChildHandle,
  killTree: async (pid) => {

    )
  }
}

function defaultInstallerPorts(fetchFn: FetchFn): InstallerPorts {
  return {
    download: async ({ url, dest, onProgress }) => {
      await downloadFile({ url, dest, fetchFn, onProgress })
    },
    extract: async (zipPath, destDir) => {
      const script = `Expand-Archive -LiteralPath '${zipPath.replace(/'/g, "''")}' -DestinationPath '${destDir.replace(/'/g, "''")}' -Force`
      await new Promise<void>((resolve, reject) =>
        execFile('powershell.exe', ['-NoProfile', '-NonInteractive', '-Command', script], { windowsHide: true }, (error) =>
          error ? reject(new Error(error.message)) : resolve()
        )
      )
    },
    validate: async (exePath) => {
      const run = (flag: string) =>
        new Promise<string>((resolve) =>
          execFile(exePath, [flag], { windowsHide: true, timeout: 15_000 }, (error, stdout, stderr) =>
            resolve(error ? '' : `${stdout}${stderr}`)
          )
        )
      const devices = await run('--list-devices')
      return devices.split(/\r?\n/).map((line) => line.trim()).filter(Boolean)
    }
  }
}
```

```ts
import type { LocalSnapshot } from '../../core/local/types'

export function useLocalState(): LocalSnapshot | null {
  const [snapshot, setSnapshot] = useState<LocalSnapshot | null>(null)

  useEffect(() => {
    let active = true
    let receivedPush = false
    const unsubscribe = window.tokenhub.local.onState((next) => {
      receivedPush = true
      setSnapshot(next)
    })
    void window.tokenhub.local
      .getState()
      .then((next) => {
        if (active && !receivedPush) setSnapshot(next)
      })
      .catch((error: unknown) => console.error('TokenHUB local state failed:', error))
    return () => {
      active = false
      unsubscribe()
    }
  }, [])

  return snapshot
}
```

```ts
import type { FetchFn } from './http'
import { quantFromFileName, splitInfo } from './library'
import type { HfModelDetail, HfModelFile, HfModelSummary } from './types'

const REPO_PATTERN = /^[\w.-]+\/[\w.-]+$/

export function hfSearchUrl(base: string, query: string): string {
  const params = new URLSearchParams({
    search: query.trim(),
    filter: 'gguf',
    sort: 'downloads',
    direction: '-1',
    limit: '50'
  })
  return `${base.replace(/\/+$/, '')}/api/models?${params.toString()}`
}

export function hfModelUrl(base: string, repo: string): string {
  return `${base.replace(/\/+$/, '')}/api/models/${repo}?blobs=true`
}

export function hfResolveUrl(base: string, repo: string, path: string): string {
  const encoded = path.split('/').map((segment) => encodeURIComponent(segment)).join('/')
  return `${base.replace(/\/+$/, '')}/${repo}/resolve/main/${encoded}`
}

function licenseOf(tags: unknown): string | null {
  if (!Array.isArray(tags)) return null
  const tag = tags.find((item) => typeof item === 'string' && item.startsWith('license:'))
  return typeof tag === 'string' ? tag.slice('license:'.length) : null
}

function gatedOf(raw: unknown): boolean {
  return raw === true || raw === 'auto' || raw === 'manual'
}

export function parseHfSearch(payload: unknown): HfModelSummary[] {
  if (!Array.isArray(payload)) return []
  return payload
    .filter((item): item is Record<string, unknown> => Boolean(item) && typeof item === 'object')
    .map((item) => ({
      id: String(item.id ?? ''),
      downloads: Number(item.downloads ?? 0),
      likes: Number(item.likes ?? 0),
      lastModified: String(item.lastModified ?? ''),
      license: licenseOf(item.tags),
      gated: gatedOf(item.gated)
    }))
    .filter((item) => item.id)
}

export function parseHfModel(repoId: string, payload: unknown): HfModelDetail {
  const data = (payload ?? {}) as Record<string, unknown>
  const siblings = Array.isArray(data.siblings) ? data.siblings : []
  const files: HfModelFile[] = []
  for (const raw of siblings) {
    if (!raw || typeof raw !== 'object') continue
    const entry = raw as { rfilename?: unknown; size?: unknown; lfs?: { size?: unknown } }
    const path = typeof entry.rfilename === 'string' ? entry.rfilename : ''
    if (!path.toLowerCase().endsWith('.gguf')) continue
    const size = Number(entry.lfs?.size ?? entry.size ?? 0)
    const name = path.split('/').pop() ?? path
    const split = splitInfo(name)
    files.push({
      path,
      sizeBytes: Number.isFinite(size) ? size : 0,
      quant: quantFromFileName(name),
      split: split ? { part: split.part, total: split.total } : null,
      isAdapter: name.toLowerCase().startsWith('mmproj')
    })
  }
  files.sort((a, b) => a.path.localeCompare(b.path))
  return { id: repoId, license: licenseOf(data.tags), gated: gatedOf(data.gated), files }
}

export interface HfClientOptions {
  fetchFn: FetchFn
  baseUrl: () => string
  token: () => string | null
}

export class HfClient {
  constructor(private readonly options: HfClientOptions) {}

  private async getJson(url: string): Promise<unknown> {
    const token = this.options.token()
    const response = await this.options.fetchFn(url, {
      headers: token ? { Authorization: `Bearer ${token}`, accept: 'application/json' } : { accept: 'application/json' }
    })
    if (response.status === 401 || response.status === 403) throw new Error('需要 HuggingFace Token（或该模型需要授权访问）')
    if (response.status === 404) throw new Error('模型不存在或已被删除')
    if (!response.ok) throw new Error(`HF 请求失败：HTTP ${response.status}`)
    return await response.json()
  }

  async search(query: string): Promise<HfModelSummary[]> {
    return parseHfSearch(await this.getJson(hfSearchUrl(this.options.baseUrl(), query)))
  }

  async model(repo: string): Promise<HfModelDetail> {
    if (!REPO_PATTERN.test(repo)) throw new Error('模型 ID 必须形如 org/repo')
    return parseHfModel(repo, await this.getJson(hfModelUrl(this.options.baseUrl(), repo)))
  }
}
```

```ts
import { mkdir, readFile, rm, writeFile } from 'node:fs/promises'
import { join } from 'node:path'
import { randomUUID } from 'node:crypto'
import { downloadFile, type FetchFn } from './http'
import { isSafeRelativePath } from './library'
import { hfResolveUrl } from './hf'
import type { DownloadFile, DownloadPhase, DownloadTask } from './types'

export interface DownloadManagerOptions {
  dataDir: string
  modelsDir: () => string
  baseUrl: () => string
  token: () => string | null
  fetchFn: FetchFn
  freeSpace?: (dir: string) => Promise<number>
  now?: () => Date
  onUpdate?: () => void
  onDone?: () => void
}

const SAFE_REPO = /^[\w.-]+\/[\w.-]+$/

export class DownloadManager {
  private tasks: DownloadTask[] = []
  private readonly aborters = new Map<string, AbortController>()
  private readonly waiters = new Map<string, Array<() => void>>()
  private running = 0
  private readonly now: () => Date
  private readonly freeSpace: (dir: string) => Promise<number>

  private constructor(private readonly options: DownloadManagerOptions) {
    this.now = options.now ?? (() => new Date())
    this.freeSpace = options.freeSpace ?? (async (dir) => {
      const { statfs } = await import('node:fs/promises')
      const info = await statfs(dir)
      return info.bavail * info.bsize
    })
  }

  static async open(options: DownloadManagerOptions): Promise<DownloadManager> {
    const manager = new DownloadManager(options)
    const text = await readFile(join(options.dataDir, 'local', 'downloads.json'), 'utf8').catch(() => null)
    if (text) {
      try {
        const parsed = JSON.parse(text) as DownloadTask[]
        manager.tasks = Array.isArray(parsed)
          ? parsed.map((task) => ({ ...task, phase: task.phase === 'done' ? 'done' : 'paused' }))
          : []
      } catch {
        manager.tasks = []
      }
    }
    return manager
  }

  getTasks(): DownloadTask[] {
    return this.tasks.map((task) => ({ ...task, files: task.files.map((file) => ({ ...file })) }))
  }

  private async persist(): Promise<void> {
    await mkdir(join(this.options.dataDir, 'local'), { recursive: true })
    await writeFile(join(this.options.dataDir, 'local', 'downloads.json'), JSON.stringify(this.tasks, null, 2), 'utf8')
  }

  private update(id: string, patch: Partial<DownloadTask>): void {
    this.tasks = this.tasks.map((task) => (task.id === id ? { ...task, ...patch, updatedAt: this.now().toISOString() } : task))
    this.options.onUpdate?.()
  }

  async enqueue(repo: string, files: string[]): Promise<void> {
    if (!SAFE_REPO.test(repo)) throw new Error('模型 ID 必须形如 org/repo')
    const normalized = files.map((file) => file.replace(/\\/g, '/'))
    for (const file of normalized) {
      if (!isSafeRelativePath(file)) throw new Error(`不安全的下载路径：${file}`)
    }
    if (normalized.length === 0) throw new Error('至少选择一个文件')
    const available = await this.freeSpace(this.options.modelsDir())
    if (available < 2 * 1024 ** 3) throw new Error('模型目录所在磁盘空间不足（至少需要 2 GB 可用）')
    const now = this.now().toISOString()
    const task: DownloadTask = {
      id: randomUUID(),
      repo,
      files: normalized.map((path) => ({ path, totalBytes: 0, receivedBytes: 0 })),
      phase: 'queued',
      error: null,
      speedBps: null,
      createdAt: now,
      updatedAt: now
    }
    this.tasks = [...this.tasks, task]
    await this.persist()
    this.options.onUpdate?.()
    void this.pump()
  }

  async pause(id: string): Promise<void> {
    this.aborters.get(id)?.abort()
    this.aborters.delete(id)
    const task = this.tasks.find((candidate) => candidate.id === id)
    if (!task || task.phase === 'done') return
    this.update(id, { phase: 'paused', speedBps: null })
    await this.persist()
  }

  async resume(id: string): Promise<void> {
    const task = this.tasks.find((candidate) => candidate.id === id)
    if (!task || task.phase === 'done') return
    this.update(id, { phase: 'queued', error: null })
    await this.persist()
    void this.pump()
  }

  async cancel(id: string): Promise<void> {
    this.aborters.get(id)?.abort()
    this.aborters.delete(id)
    const task = this.tasks.find((candidate) => candidate.id === id)
    if (!task) return
    for (const file of task.files) {
      if (file.receivedBytes > 0) {
        await rm(join(this.options.modelsDir(), task.repo, `${file.path}.part`), { force: true })
      }
    }
    this.update(id, { phase: 'canceled', speedBps: null })
    await this.persist()
  }

  async waitForIdle(): Promise<void> {
    while (this.running > 0 || this.tasks.some((task) => task.phase === 'queued')) {
      await new Promise((resolve) => setTimeout(resolve, 5))
    }
  }

  private async pump(): Promise<void> {
    if (this.running >= 2) return
    const next = this.tasks.find((task) => task.phase === 'queued')
    if (!next) return
    this.running += 1
    void this.run(next.id).finally(() => {
      this.running -= 1
      void this.pump()
    })
  }

  private async run(id: string): Promise<void> {
    const task = this.tasks.find((candidate) => candidate.id === id)
    if (!task) return
    this.update(id, { phase: 'downloading', error: null })
    const controller = new AbortController()
    this.aborters.set(id, controller)
    try {
      for (const file of task.files) {
        const dest = join(this.options.modelsDir(), task.repo, file.path)
        const token = this.options.token()
        let lastAt = Date.now()
        let lastBytes = file.receivedBytes
        await downloadFile({
          url: hfResolveUrl(this.options.baseUrl(), task.repo, file.path),
          dest,
          headers: token ? { Authorization: `Bearer ${token}` } : undefined,
          fetchFn: this.options.fetchFn,
          signal: controller.signal,
          onProgress: (received, total) => {
            const now = Date.now()
            const speed = now - lastAt >= 1_000 ? ((received - lastBytes) * 1000) / (now - lastAt) : null
            if (speed !== null) {
              lastAt = now
              lastBytes = received
            }
            this.tasks = this.tasks.map((candidate) =>
              candidate.id === id
                ? {
                    ...candidate,
                    files: candidate.files.map((item) =>
                      item.path === file.path ? { ...item, receivedBytes: received, totalBytes: total ?? item.totalBytes } : item
                    ),
                    speedBps: speed ?? candidate.speedBps
                  }
                : candidate
            )
            this.options.onUpdate?.()
          }
        })
      }
      this.update(id, { phase: 'done', speedBps: null })
      this.options.onDone?.()
    } catch (error) {
      if (controller.signal.aborted) {
        this.update(id, { phase: 'paused', speedBps: null })
      } else {
        this.update(id, { phase: 'error', error: error instanceof Error ? error.message : String(error), speedBps: null })
      }
    } finally {
      this.aborters.delete(id)
      await this.persist()
    }
  }

  async shutdown(): Promise<void> {
    for (const aborter of this.aborters.values()) aborter.abort()
    this.aborters.clear()
    await this.persist()
  }
}
```

```tsx
import { useState } from 'react'
import type { HfModelDetail, HfModelSummary } from '../../../core/local/types'
import { formatNumber, errorMessage } from '../../lib/format'

function gb(bytes: number): string {
  return `${(bytes / 1024 ** 3).toFixed(2)} GB`
}

export function HfSearchPanel({ onDownloaded, onError }: { onDownloaded: () => void; onError: (message: string) => void }) {
  const [query, setQuery] = useState('')
  const [results, setResults] = useState<HfModelSummary[]>([])
  const [detail, setDetail] = useState<HfModelDetail | null>(null)
  const [selected, setSelected] = useState<string[]>([])
  const [busy, setBusy] = useState(false)

  async function search() {
    setBusy(true)
    try {
      setResults(await window.tokenhub.local.searchHf(query))
      setDetail(null)
    } catch (error) {
      onError(`搜索失败：${errorMessage(error)}`)
    } finally {
      setBusy(false)
    }
  }

  async function open(repo: string) {
    setBusy(true)
    try {
      const model = await window.tokenhub.local.getHfModel(repo)
      setDetail(model)
      setSelected(model.files.filter((file) => !file.isAdapter).map((file) => file.path))
    } catch (error) {
      onError(`读取模型失败：${errorMessage(error)}`)
    } finally {
      setBusy(false)
    }
  }

  async function download() {
    if (!detail || selected.length === 0) return
    try {
      await window.tokenhub.local.download(detail.id, selected)
      onDownloaded()
    } catch (error) {
      onError(`开始下载失败：${errorMessage(error)}`)
    }
  }

  return (
    <section className="mt-4 rounded-lg border border-line bg-surface p-5 shadow-sm">
      <h2 className="text-sm font-semibold text-ink">HuggingFace 模型</h2>
      <div className="mt-3 flex gap-2">
        <input
          value={query}
          onChange={(event) => setQuery(event.target.value)}
          onKeyDown={(event) => {
            if (event.key === 'Enter') void search()
          }}
          placeholder="搜索 GGUF 模型，如 qwen3 gguf"
          className="h-9 w-full rounded-lg border border-line bg-surface px-3 text-sm text-ink focus:border-accent focus:outline-none"
        />
        <button type="button" onClick={() => void search()} disabled={busy} className="inline-flex h-9 shrink-0 items-center rounded-lg bg-accent px-3.5 text-sm font-medium text-white disabled:opacity-60">
          {busy ? '查询中…' : '搜索'}
        </button>
      </div>
      <ul className="mt-3 max-h-64 divide-y divide-line overflow-auto">
        {results.map((model) => (
          <li key={model.id} className="flex items-center justify-between gap-3 py-2">
            <button type="button" onClick={() => void open(model.id)} className="min-w-0 text-left">
              <p className="truncate text-sm text-ink">{model.id}</p>
              <p className="text-xs text-muted">
                下载 {formatNumber(model.downloads)} · 赞 {formatNumber(model.likes)}
                {model.license ? ` · ${model.license}` : ''}
                {model.gated ? ' · 需授权' : ''}
              </p>
            </button>
          </li>
        ))}
        {results.length === 0 ? <li className="py-3 text-sm text-muted">输入关键词搜索 GGUF 模型。</li> : null}
      </ul>
      {detail ? (
        <div className="mt-4 rounded-md bg-canvas p-3">
          <p className="text-sm font-medium text-ink">{detail.id}</p>
          <ul className="mt-2 space-y-1">
            {detail.files.map((file) => (
              <li key={file.path} className="flex items-center gap-2 text-xs text-muted">
                <input
                  type="checkbox"
                  checked={selected.includes(file.path)}
                  onChange={(event) =>
                    setSelected((current) =>
                      event.target.checked ? [...current, file.path] : current.filter((path) => path !== file.path)
                    )
                  }
                />
                <span className="min-w-0 flex-1 truncate">
                  {file.path}
                  {file.split ? `（分片 ${file.split.part}/${file.split.total}）` : ''}
                  {file.isAdapter ? '（视觉适配，P3）' : ''}
                </span>
                <span className="font-mono">{gb(file.sizeBytes)}</span>
              </li>
            ))}
          </ul>
          <button
            type="button"
```

```tsx
import { errorMessage } from '../../lib/format'
import type { DownloadTask } from '../../../core/local/types'

function percent(task: DownloadTask): number {
  const total = task.files.reduce((sum, file) => sum + file.totalBytes, 0)
  const received = task.files.reduce((sum, file) => sum + file.receivedBytes, 0)
  return total > 0 ? Math.min(100, Math.round((received / total) * 100)) : 0
}

export function DownloadQueue({ tasks, onError }: { tasks: DownloadTask[]; onError: (message: string) => void }) {
  async function act(action: 'pause' | 'resume' | 'cancel', id: string) {
    try {
      if (action === 'pause') await window.tokenhub.local.pauseDownload(id)
      else if (action === 'resume') await window.tokenhub.local.resumeDownload(id)
      else await window.tokenhub.local.cancelDownload(id)
    } catch (error) {
      onError(`操作失败：${errorMessage(error)}`)
    }
  }
  return (
    <section className="mt-4 rounded-lg border border-line bg-surface p-5 shadow-sm">
      <h2 className="text-sm font-semibold text-ink">下载队列</h2>
      <ul className="mt-3 space-y-3">
        {tasks.map((task) => (
          <li key={task.id}>
            <div className="flex items-center justify-between gap-3 text-xs">
              <span className="min-w-0 truncate text-sm text-ink">{task.repo}</span>
              <span className="text-muted">
                {task.phase}
                {task.speedBps ? ` · ${(task.speedBps / 1024 ** 2).toFixed(1)} MB/s` : ''}
              </span>
            </div>
            <div className="mt-1 h-1.5 overflow-hidden rounded-full bg-meter-track">
              <div className="h-full bg-meter-primary" style={{ width: `${percent(task)}%` }} />
            </div>
            {task.error ? <p className="mt-1 text-xs text-red-600">{task.error}</p> : null}
            <div className="mt-1 flex gap-2 text-xs">
              {task.phase === 'downloading' ? (
                <button type="button" onClick={() => void act('pause', task.id)} className="text-muted hover:text-ink">暂停</button>
              ) : null}
              {task.phase === 'paused' || task.phase === 'error' ? (
                <button type="button" onClick={() => void act('resume', task.id)} className="text-muted hover:text-ink">继续</button>
              ) : null}
              {task.phase !== 'done' && task.phase !== 'canceled' ? (
                <button type="button" onClick={() => void act('cancel', task.id)} className="text-red-600 hover:text-red-700">取消</button>
              ) : null}
            </div>
          </li>
        ))}
        {tasks.length === 0 ? <li className="text-sm text-muted">暂无下载任务。</li> : null}
      </ul>
    </section>
  )
}
```
