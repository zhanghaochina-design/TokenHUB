# TokenHUB Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在 Windows 上交付一个 Electron 桌面应用，把 OpenCode Go、CommandCode Goat、DeepSeek 三个 OpenAI 兼容供应商合并成本地双协议 API 网关，按手动优先级路由并在额度耗尽时自动切换。

**Architecture:** 单进程 Electron；主进程内嵌 Hono 服务（`127.0.0.1:8787`）和全部核心逻辑，渲染进程只通过类型化 IPC 操作配置与查看状态。核心层（`core/`）不依赖 Electron，可独立测试。

**Tech Stack:** Electron 38+、TypeScript strict、electron-vite、React 19、Tailwind CSS v4、Hono、@hono/node-server、undici、Vitest、Playwright（Electron）、electron-builder。

**Spec:** `docs/superpowers/specs/2026-10-28-tokenhub-design.md`

## Global Constraints

- 目标平台 Windows 11 x64；代码不引入 Linux/macOS 专属依赖。
- 本地服务只绑定 `127.0.0.1`，默认端口 `8787`；只允许 `https://` 上游，例外：`http://127.0.0.1`（本地模型）。
- 供应商 baseURL 预设：OpenCode Go `https://opencode.ai/zen/go/v1`；CommandCode `https://api.commandcode.ai/provider/v1`；DeepSeek `https://api.deepseek.com/v1`。
- 额度查询：OpenCode Go `GET https://opencode.ai/zen/go/v1/usage`；DeepSeek `GET https://api.deepseek.com/user/balance`；CommandCode 无探针（错误码模式）。
- 额度错误关键词：`quota|insufficient|balance|billing|payment|额度|余额|欠费|上限`（不要用裸 `limit`，否则普通限流会被误判）。
- 超时：连接 15s、响应头 120s、流式闲置 120s（由 undici Agent 统一配置）。
- 请求体上限 20MB；历史只存元数据；Key 与请求/响应正文永不写日志或历史。
- 默认冷却：额度耗尽 5 小时（可配），普通限流 60 秒或 `Retry-After`，瞬时故障 60 秒。
- 历史保留 30 天（可配 7/30/90/永久）。
- **项目规则覆盖计划模板：未经用户明确要求不执行 `git commit` / `git push`。每个任务以“测试通过 + 类型检查 + `git status`”为检查点，提交由用户统一决定。**
- 代码注释只写非显然的 WHY 或契约，不写变更史。

```ts
import { mkdtemp, readFile } from 'node:fs/promises'
import { tmpdir } from 'node:os'
import { join } from 'node:path'
import { EncryptionUnavailableError, SecretStore, type CipherPort } from '../../core/store/secrets'

const xorCipher: CipherPort = {
  isAvailable: () => true,
  encrypt: (s) => Buffer.from([...Buffer.from(s, 'utf8')].map((b) => b ^ 0x5a)),
  decrypt: (b) => Buffer.from([...b].map((x) => x ^ 0x5a)).toString('utf8')

async function tempFile(): Promise<string> {
  return join(await mkdtemp(join(tmpdir(), 'tokenhub-secrets-')), 'secrets.bin')
}

describe('secret store', () => {
  it('round-trips provider keys without plaintext on disk', async () => {
    const file = await tempFile()
    const store = await SecretStore.open(file, xorCipher)
    await store.setProviderKey('p1', 'sk-super-secret')
    await store.setLocalApiKey('local-key')
    const reloaded = await SecretStore.open(file, xorCipher)
    expect(reloaded.getProviderKey('p1')).toBe('sk-super-secret')
    expect(reloaded.getLocalApiKey()).toBe('local-key')
    expect(await readFile(file, 'utf8')).not.toContain('sk-super-secret')

    const file = await tempFile()

    await store.removeProviderKey('p1')
```

```ts
import { mkdir, readFile, rename, rm, writeFile } from 'node:fs/promises'
import { dirname } from 'node:path'

export async function readText(file: string): Promise<string | null> {
  try {
    return await readFile(file, 'utf8')
  } catch {
    return null
  }
}

export async function writeFileAtomic(file: string, content: string): Promise<void> {
  await mkdir(dirname(file), { recursive: true })
  const tmp = `${file}.tmp`
  await writeFile(tmp, content, 'utf8')
  try {
    await rename(tmp, file)
  } catch {
    await rm(file, { force: true })
    await rename(tmp, file)
  }
}
```

```ts
export function normalizeConfig(raw: unknown): AppConfig {
  const input = (raw ?? {}) as Partial<AppConfig>
  const base = defaultConfig()
  const settings = { ...base.settings, ...(input.settings ?? {}) }
  const providers = Array.isArray(input.providers) ? input.providers.map(normalizeProvider) : []
  return { version: 1, providers, settings }
}

function normalizeProvider(raw: unknown): ProviderConfig {
  const p = (raw ?? {}) as Partial<ProviderConfig>
  return {
    id: String(p.id ?? crypto.randomUUID()),
    name: String(p.name ?? 'Unnamed'),
    preset: p.preset ?? 'custom',
    baseUrl: String(p.baseUrl ?? '').replace(/\/+$/, ''),
    quotaAdapter: p.quotaAdapter ?? 'none',
    enabled: p.enabled !== false,
    models: {
      cached: Array.isArray(p.models?.cached) ? p.models.cached.map(String) : [],
```

```ts
const MAGIC = 'TOKENHUB-SECRETS-1'

export interface CipherPort {
  isAvailable(): boolean
  encrypt(plain: string): Buffer
  decrypt(data: Buffer): string
}

export class EncryptionUnavailableError extends Error {
  constructor() {
    super('系统加密不可用（safeStorage/DPAPI），拒绝以明文保存密钥')
  }
}

interface Secrets {
  version: 1
  providerKeys: Record<string, string>
  localApiKey: string | null
}

function emptySecrets(): Secrets {
  return { version: 1, providerKeys: {}, localApiKey: null }
}

export class SecretStore {
  private data: Secrets = emptySecrets()

  private constructor(private readonly file: string, private readonly cipher: CipherPort) {}

  static async open(file: string, cipher: CipherPort): Promise<SecretStore> {
    const store = new SecretStore(file, cipher)
    const text = await readText(file)
    if (text) {
      const [magic, payload] = text.split('\n')
      if (magic === MAGIC && payload) store.data = JSON.parse(cipher.decrypt(Buffer.from(payload, 'base64'))) as Secrets
    }
    return store
  }

  getProviderKey(id: string): string | null {
    return this.data.providerKeys[id] ?? null
  }

  hasProviderKey(id: string): boolean {
    return Boolean(this.data.providerKeys[id])
  }

  getLocalApiKey(): string | null {
    return this.data.localApiKey
  }

  async setProviderKey(id: string, key: string): Promise<void> {
    this.data.providerKeys[id] = key
    await this.persist()
  }

  async removeProviderKey(id: string): Promise<void> {
    delete this.data.providerKeys[id]
    await this.persist()
  }

  async setLocalApiKey(key: string | null): Promise<void> {
    this.data.localApiKey = key
    await this.persist()
  }

  private async persist(): Promise<void> {
    if (!this.cipher.isAvailable()) throw new EncryptionUnavailableError()
    const payload = this.cipher.encrypt(JSON.stringify(this.data)).toString('base64')
    await writeFileAtomic(this.file, `${MAGIC}\n${payload}`)
  }
}
```

```ts
import type { MergedModel, ProviderConfig } from '../types'

export function providerSupports(provider: ProviderConfig, model: string): boolean {
  return provider.models.cached.includes(model) || provider.models.manual.includes(model)
}

export function buildCatalog(providers: ProviderConfig[]): MergedModel[] {
  const map = new Map<string, string[]>()
  for (const provider of providers) {
    if (!provider.enabled) continue
    for (const model of new Set([...provider.models.cached, ...provider.models.manual])) {
      map.set(model, [...(map.get(model) ?? []), provider.id])
    }
  }
  return [...map.entries()].map(([id, ids]) => ({ id, providers: ids })).sort((a, b) => a.id.localeCompare(b.id))
}

export function parseRequestedModel(raw: string, providers: ProviderConfig[]): { providerId: string | null; model: string } {
  const slash = raw.indexOf('/')
  if (slash > 0) {
    const prefix = raw.slice(0, slash)
    if (providers.some((p) => p.id === prefix)) return { providerId: prefix, model: raw.slice(slash + 1) }
  }
  return { providerId: null, model: raw }
}
```

```ts
import type { UpstreamFailure } from './types'

const QUOTA_RE = /quota|insufficient|balance|billing|payment|额度|余额|欠费|上限/i

export function parseRetryAfter(raw: string | null, now: number = Date.now()): number | null {
  if (!raw) return null
  const seconds = Number(raw)
  if (Number.isFinite(seconds)) return Math.max(0, seconds * 1000)
  const date = Date.parse(raw)
  return Number.isNaN(date) ? null : Math.max(0, date - now)
}

function messageOf(bodyText: string): string {
  try {
    const parsed = JSON.parse(bodyText) as { error?: { message?: string } | string; message?: string }
    if (typeof parsed.error === 'string') return parsed.error
    return parsed.error?.message ?? parsed.message ?? bodyText.slice(0, 500)
  } catch {
    return bodyText.slice(0, 500)
  }
}

export function classifyUpstreamResponse(status: number, bodyText: string, headers: Headers): UpstreamFailure {
  const message = messageOf(bodyText)
  const retryAfterMs = parseRetryAfter(headers.get('retry-after'))
  if (status === 402) return { kind: 'quota', status, message, retryAfterMs }
  if (status === 429) return { kind: QUOTA_RE.test(bodyText) ? 'quota' : 'rate_limit', status, message, retryAfterMs }
  if (status === 401) return { kind: 'auth', status, message, retryAfterMs }
  if (status === 403) return { kind: QUOTA_RE.test(bodyText) ? 'quota' : 'auth', status, message, retryAfterMs }
  if (status === 404) return { kind: 'model_missing', status, message, retryAfterMs }
  if (status === 400 || status === 422) return { kind: 'request', status, message, retryAfterMs }
  if (status >= 500) return { kind: 'transient', status, message, retryAfterMs }
  return { kind: 'unknown', status, message, retryAfterMs }
}
```

```ts
import type { QuotaAdapter, ProbeOptions } from './types'
import type { ProviderConfig, QuotaSnapshot, QuotaWindow } from '../types'

const USAGE_URL = 'https://opencode.ai/zen/go/v1/usage'

function toWindow(value: unknown): QuotaWindow | undefined {
  if (!value || typeof value !== 'object') return undefined
  const v = value as { percent?: unknown; status?: unknown; resetsAt?: unknown }
  if (typeof v.percent !== 'number') return undefined
  return {
    percent: v.percent,
    status: typeof v.status === 'string' ? v.status : 'ok',
    resetsAt: typeof v.resetsAt === 'string' ? v.resetsAt : null
  }
}

export const opencodeGoAdapter: QuotaAdapter = {
  id: 'opencode-go',
  async probe(provider: ProviderConfig, apiKey: string, options: ProbeOptions = {}): Promise<QuotaSnapshot> {
    const now = options.now ?? (() => new Date())
    const fetchFn = options.fetchFn ?? fetch
    const res = await fetchFn(USAGE_URL, {
      headers: { Authorization: `Bearer ${apiKey}`, Accept: 'application/json' },
      signal: AbortSignal.timeout(15_000)
    })
    const fetchedAt = now().toISOString()
    if (res.status === 401) {
      return { providerId: provider.id, availability: 'auth_error', source: 'probe', fetchedAt, note: 'usage 401' }
    }
    if (!res.ok) throw new Error(`usage ${res.status}`)
    const json = (await res.json()) as { usage?: Record<string, unknown> }
    const windows = {
      rolling: toWindow(json.usage?.rolling),
      weekly: toWindow(json.usage?.weekly),
      monthly: toWindow(json.usage?.monthly)
    }
    const list = [windows.rolling, windows.weekly, windows.monthly].filter((w): w is QuotaWindow => Boolean(w))
    const exhausted = list.some((w) => w.percent >= 100 || w.status === 'rate-limited')
    return {
      providerId: provider.id,
      availability: exhausted ? 'exhausted' : 'ok',
      windows,
      source: 'probe',
      fetchedAt
    }
  }
}
```

```ts
import type { ProviderConfig, ProviderRuntime, QuotaSnapshot, UpstreamFailure } from '../types'

const TRANSIENT_COOLDOWN_MS = 60_000

export class RuntimeStore {
  private states = new Map<string, ProviderRuntime>()

  constructor(private readonly now: () => Date = () => new Date()) {}

  get(id: string): ProviderRuntime {
    let state = this.states.get(id)
    if (!state) {
      state = {
        providerId: id,
        availability: 'unknown',
        cooldownUntil: null,
        lastErrorKind: null,
        consecutiveFailures: 0,
        quota: null,
        probeFailures: 0,
        probeDisabled: false
      }
      this.states.set(id, state)
    }
    return state
  }

  all(): ProviderRuntime[] {
    return [...this.states.values()]
  }

  isRoutable(id: string): boolean {
    const state = this.get(id)
    if (state.availability === 'auth_error') return false
    if (state.cooldownUntil && Date.parse(state.cooldownUntil) > this.now().getTime()) return false
    return true
  }

  markSuccess(id: string): void {
    const state = this.get(id)
    state.availability = 'ok'
    state.cooldownUntil = null
    state.lastErrorKind = null
    state.consecutiveFailures = 0
  }

  markFailure(provider: ProviderConfig, failure: UpstreamFailure): void {
    const state = this.get(provider.id)
    state.lastErrorKind = failure.kind
    state.consecutiveFailures += 1
    const now = this.now().getTime()
    if (failure.kind === 'quota') {
      state.availability = 'exhausted'
      state.cooldownUntil = new Date(now + provider.quota.quotaCooldownMs).toISOString()
    } else if (failure.kind === 'rate_limit') {
      state.availability = 'rate_limited'
      state.cooldownUntil = new Date(now + (failure.retryAfterMs ?? TRANSIENT_COOLDOWN_MS)).toISOString()
    } else if (failure.kind === 'auth') {
      state.availability = 'auth_error'
      state.cooldownUntil = null
    } else if (failure.kind === 'transient') {
      state.availability = 'unknown'
      state.cooldownUntil = new Date(now + TRANSIENT_COOLDOWN_MS).toISOString()
    }
  }

  markProbe(provider: ProviderConfig, snapshot: QuotaSnapshot): void {
    const state = this.get(provider.id)
    state.quota = snapshot
    if (snapshot.availability === 'auth_error') {
      state.availability = 'auth_error'
      state.cooldownUntil = null
      return
    }
    if (snapshot.availability === 'exhausted') {
      state.availability = 'exhausted'
      const windows = [snapshot.windows?.rolling, snapshot.windows?.weekly, snapshot.windows?.monthly]
      const resets = windows
        .filter((w) => w && (w.percent >= 100 || w.status === 'rate-limited'))
        .map((w) => w!.resetsAt)
        .filter((v): v is string => Boolean(v))
      const latest = resets.sort().at(-1)
      const until = latest ? Date.parse(latest) : this.now().getTime() + provider.quota.quotaCooldownMs
      state.cooldownUntil = new Date(Math.max(until, this.now().getTime())).toISOString()
      return
    }
    state.availability = 'ok'
    state.cooldownUntil = null
    state.lastErrorKind = null
  }

  reset(id: string): void {
    const state = this.get(id)
    state.availability = 'ok'
    state.cooldownUntil = null
    state.lastErrorKind = null
    state.consecutiveFailures = 0
    state.probeFailures = 0
    state.probeDisabled = false
  }

  remove(id: string): void {
    this.states.delete(id)
  }
}
```

```ts
import type { MergedModel, ProviderConfig, UpstreamFailure } from '../types'
import { parseRequestedModel, providerSupports } from '../providers/catalog'
import type { RuntimeStore } from './runtime'

export interface Candidate {
  provider: ProviderConfig
  upstreamModel: string
}

export interface Resolution {
  model: string
  modelKnown: boolean
  candidates: Candidate[]
}

export function resolveCandidates(input: {
  requestedModel: string
  providers: ProviderConfig[]
  catalog: MergedModel[]
  runtime: RuntimeStore
}): Resolution {
  const { requestedModel, providers, catalog, runtime } = input
  const parsed = parseRequestedModel(requestedModel, providers)
  if (parsed.providerId) {
    const provider = providers.find((p) => p.id === parsed.providerId)!
    const known = provider.enabled && providerSupports(provider, parsed.model)
    const candidates = known && runtime.isRoutable(provider.id) ? [{ provider, upstreamModel: parsed.model }] : []
    return { model: parsed.model, modelKnown: known, candidates }
  }
  const entry = catalog.find((m) => m.id === parsed.model)
  if (!entry) return { model: parsed.model, modelKnown: false, candidates: [] }
  const candidates: Candidate[] = []
  for (const id of entry.providers) {
    const provider = providers.find((p) => p.id === id)
    if (!provider || !provider.enabled || !providerSupports(provider, parsed.model) || !runtime.isRoutable(id)) continue
    candidates.push({ provider, upstreamModel: parsed.model })
  }
  return { model: parsed.model, modelKnown: true, candidates }
}

export type AttemptResult<T> = { ok: true; value: T } | { ok: false; failure: UpstreamFailure }

export type FailoverResult<T> =
  | { ok: true; candidate: Candidate; value: T; tried: string[] }
  | { ok: false; failure: UpstreamFailure; tried: string[] }

export async function executeWithFailover<T>(
  candidates: Candidate[],
  attempt: (candidate: Candidate) => Promise<AttemptResult<T>>,
  runtime: RuntimeStore,
  hooks: { onModelMissing?: (providerId: string) => void; onQuotaFailure?: (providerId: string) => void } = {}
): Promise<FailoverResult<T>> {
  const tried: string[] = []
  let lastFailure: UpstreamFailure | null = null
  for (const candidate of candidates) {
    tried.push(candidate.provider.id)
    const result = await attempt(candidate)
    if (result.ok) {
      runtime.markSuccess(candidate.provider.id)
      return { ok: true, candidate, value: result.value, tried }
    }
    lastFailure = result.failure
    if (result.failure.kind === 'model_missing') hooks.onModelMissing?.(candidate.provider.id)
    if (result.failure.kind === 'quota') hooks.onQuotaFailure?.(candidate.provider.id)
    runtime.markFailure(candidate.provider, result.failure)
    if (result.failure.kind === 'request') break
  }
  return {
    ok: false,
    failure: lastFailure ?? { kind: 'unknown', status: null, message: 'no candidate providers', retryAfterMs: null },
    tried
  }
}
```

```ts
  handler: (path: string, body: string) => FakeResponse | Promise<FakeResponse>

    let body = ''

    requests.push({ path: req.url ?? '', body })

    res.writeHead(out.status, { 'content-type': 'application/json', ...(out.headers ?? {}) })
    res.end(out.body ?? '')
  })
  await new Promise<void>((resolve) => server.listen(0, '127.0.0.1', resolve))
  const address = server.address()
  const port = typeof address === 'object' && address ? address.port : 0
  return {
    url: `http://127.0.0.1:${port}`,
    requests,
    close: () => new Promise((resolve) => server.close(() => resolve()))
  }
```

```ts
import { Agent, fetch as undiciFetch } from 'undici'
import type { ProviderConfig, UpstreamFailure } from '../types'
import { classifyUpstreamResponse } from '../errors'
import { CONNECT_TIMEOUT_MS, FIRST_BYTE_TIMEOUT_MS, STREAM_IDLE_TIMEOUT_MS } from '../appInfo'

export type FetchFn = typeof undiciFetch

const agent = new Agent({
  connect: { timeout: CONNECT_TIMEOUT_MS },
  headersTimeout: FIRST_BYTE_TIMEOUT_MS,
  bodyTimeout: STREAM_IDLE_TIMEOUT_MS
})

export interface ForwardOptions {
  provider: ProviderConfig
  upstreamModel: string
  apiKey: string
  body: Record<string, unknown>
  signal: AbortSignal
  fetchFn?: FetchFn
  onDisableStreamUsage?: (providerId: string) => void
}

export type ForwardOutcome =
  | { ok: true; response: Response; injectedStreamUsage: boolean }
  | { ok: false; failure: UpstreamFailure }

export async function forwardOpenAI(options: ForwardOptions): Promise<ForwardOutcome> {
  const { provider, upstreamModel, apiKey, body, signal, onDisableStreamUsage } = options
  const fetchFn = options.fetchFn ?? undiciFetch
  const wantsStream = body.stream === true
  const inject = wantsStream && provider.quota.injectStreamUsage
  const payload: Record<string, unknown> = { ...body, model: upstreamModel }
  if (inject) payload.stream_options = { ...(body.stream_options as object | undefined), include_usage: true }

  const send = async (withInject: boolean): Promise<Response> =>
    fetchFn(`${provider.baseUrl.replace(/\/+$/, '')}/chat/completions`, {
      method: 'POST',
      headers: {
        Authorization: `Bearer ${apiKey}`,
        'Content-Type': 'application/json',
        Accept: withInject || wantsStream ? 'text/event-stream' : 'application/json'
      },
      body: JSON.stringify(payload),
      signal,
      dispatcher: agent
    } as never)

  let response: Response
  try {
    response = await send(inject)
  } catch (error) {
    return {
      ok: false,
      failure: { kind: 'transient', status: null, message: error instanceof Error ? error.message : 'network error', retryAfterMs: null }
    }
  }

  if (!response.ok) {
    const text = (await response.text().catch(() => '')).slice(0, 8_000)
    if (inject && response.status === 400 && /stream_options/i.test(text)) {
      onDisableStreamUsage?.(provider.id)
      payload.stream_options = body.stream_options
      try {
        response = await send(false)
      } catch (error) {
        return {
          ok: false,
          failure: { kind: 'transient', status: null, message: error instanceof Error ? error.message : 'network error', retryAfterMs: null }
        }
      }
      if (!response.ok) {
        const retryText = (await response.text().catch(() => '')).slice(0, 8_000)
        return { ok: false, failure: classifyUpstreamResponse(response.status, retryText, response.headers) }
      }
      return { ok: true, response, injectedStreamUsage: false }
    }
    return { ok: false, failure: classifyUpstreamResponse(response.status, text, response.headers) }
  }

  return { ok: true, response, injectedStreamUsage: inject }
}
```

```ts
import { Hono } from 'hono'
import { bodyLimit } from 'hono/body-limit'
import type { AppConfig, HistorySink, MergedModel } from '../types'
import type { RuntimeStore } from '../router/runtime'
import { resolveCandidates, executeWithFailover } from '../router/router'
import { openAIError, protocolError } from './errors'
import { providedKey, safeEqual } from './auth'
import { forwardOpenAI } from './forward'
import { MAX_BODY_BYTES } from '../appInfo'

export interface AppDeps {
  getConfig(): AppConfig
  getProviderKey(id: string): string | null
  getLocalApiKey(): string | null
  getCatalog(): MergedModel[]
  runtime: RuntimeStore
  history: HistorySink
  isPaused(): boolean
  getPort?(): number
  onModelMissing?(providerId: string): void
  onQuotaFailure?(providerId: string): void
  onDisableStreamUsage?(providerId: string): void
  now?: () => Date
  fetchFn?: Parameters<typeof forwardOpenAI>[0]['fetchFn']
}

export function createApp(deps: AppDeps): Hono {
  const app = new Hono()
  const now = deps.now ?? (() => new Date())

  app.use('*', bodyLimit({
    maxSize: MAX_BODY_BYTES,
    onError: (c) => protocolError(c, 413, 'request body too large')
  }))

  app.use('/v1/*', async (c, next) => {
    const localKey = deps.getLocalApiKey()
    if (localKey) {
      const key = providedKey(c.req.raw.headers)
      if (!key || !safeEqual(key, localKey)) {
        return protocolError(c, 401, 'invalid local api key', 'authentication_error')
      }
    }
    if (deps.isPaused()) return protocolError(c, 503, 'routing is paused')
    await next()
  })

  app.get('/health', (c) => c.json({ status: 'ok', name: 'TokenHUB', port: deps.getPort?.() ?? null, paused: deps.isPaused() }))

  app.get('/v1/models', (c) => {
    const catalog = deps.getCatalog()
    if (c.req.header('anthropic-version')) {
```

```ts
export interface SSEEvent {
  event?: string
  data: string
}

export async function* parseSSE(stream: ReadableStream<Uint8Array>): AsyncGenerator<SSEEvent> {
  const reader = stream.getReader()
  const decoder = new TextDecoder()
  let buffer = ''
  while (true) {
    const { done, value } = await reader.read()
    if (done) break
    buffer += decoder.decode(value, { stream: true }).replace(/\r\n/g, '\n')
    let idx: number
    while ((idx = buffer.indexOf('\n\n')) >= 0) {
      const raw = buffer.slice(0, idx)
      buffer = buffer.slice(idx + 2)
      let event: string | undefined
      const data: string[] = []
      for (const line of raw.split('\n')) {
        if (line.startsWith('event:')) event = line.slice(6).trim()
        else if (line.startsWith('data:')) data.push(line.slice(5).trimStart())
      }
      if (data.length) yield { event, data: data.join('\n') }
    }
  }
  if (buffer.trim()) {
    const data = buffer.split('\n').filter((l) => l.startsWith('data:')).map((l) => l.slice(5).trimStart())
    if (data.length) yield { data: data.join('\n') }
  }
}

export interface TapHooks {
  onUsage?(usage: { prompt: number; completion: number }): void
  onDone?(): void
  onError?(error: unknown): void
}

export function tapSSEUsage(stream: ReadableStream<Uint8Array>, hooks: TapHooks): ReadableStream<Uint8Array> {
  const [forTap, forClient] = stream.tee()
  void (async () => {
    try {
      for await (const event of parseSSE(forTap)) {
        if (event.data === '[DONE]') break
        try {
          const parsed = JSON.parse(event.data) as { usage?: { prompt_tokens?: number; completion_tokens?: number } }
          if (parsed.usage) {
            hooks.onUsage?.({
              prompt: parsed.usage.prompt_tokens ?? 0,
              completion: parsed.usage.completion_tokens ?? 0
            })
          }
        } catch {
          // 非 JSON 注释帧直接跳过
        }
      }
    } catch (error) {
      hooks.onError?.(error)
    } finally {
      hooks.onDone?.()
    }
  })()
  return forClient
}
```

```ts
import { appendFile, mkdir, readdir, readFile, unlink } from 'node:fs/promises'
import { join } from 'node:path'
import type { HistoryRecord, HistorySink, TodayStats } from '../types'

function dayKey(date: Date): string {
  const y = date.getFullYear()
  const m = String(date.getMonth() + 1).padStart(2, '0')
  const d = String(date.getDate()).padStart(2, '0')
  return `${y}-${m}-${d}`
}

export class HistoryStore implements HistorySink {
  constructor(private readonly dir: string, private readonly now: () => Date = () => new Date()) {}

  async append(record: HistoryRecord): Promise<void> {
    await mkdir(this.dir, { recursive: true })
    const day = dayKey(new Date(record.ts))
    await appendFile(join(this.dir, `${day}.jsonl`), `${JSON.stringify(record)}\n`, 'utf8')
  }

  async list(day: string, limit = 200): Promise<HistoryRecord[]> {
    try {
      const text = await readFile(join(this.dir, `${day}.jsonl`), 'utf8')
      const rows = text.split('\n').filter(Boolean).map((line) => JSON.parse(line) as HistoryRecord)
      return rows.reverse().slice(0, limit)
    } catch {
      return []
    }
  }

  async statsFor(day: string): Promise<TodayStats> {
    const rows = await this.list(day, Number.MAX_SAFE_INTEGER)
    return {
      requests: rows.length,
      promptTokens: rows.reduce((sum, r) => sum + (r.promptTokens ?? 0), 0),
      completionTokens: rows.reduce((sum, r) => sum + (r.completionTokens ?? 0), 0),
      failures: rows.filter((r) => r.outcome === 'error').length
    }
  }

  async prune(retentionDays: number): Promise<void> {
    if (retentionDays <= 0) return
    const files = await readdir(this.dir).catch(() => [] as string[])
    const cutoff = dayKey(new Date(this.now().getTime() - retentionDays * 86_400_000))
    for (const file of files) {
      if (/^\d{4}-\d{2}-\d{2}\.jsonl$/.test(file) && file.slice(0, 10) < cutoff) {
        await unlink(join(this.dir, file)).catch(() => {})
      }
    }
  }

  today(): string {
    return dayKey(this.now())
  }
}
```

```ts
import { appendFile, mkdir, rename, stat } from 'node:fs/promises'
import { dirname } from 'node:path'

const MAX_BYTES = 5 * 1024 * 1024
const KEEP = 5

export class Logger {
  constructor(private readonly file: string) {}

  async info(message: string): Promise<void> {
    await this.write('INFO', message)
  }

  async error(message: string): Promise<void> {
    await this.write('ERROR', message)
  }

  private async write(level: string, message: string): Promise<void> {
    try {
      await mkdir(dirname(this.file), { recursive: true })
      await this.rotateIfNeeded()
      await appendFile(this.file, `${new Date().toISOString()} ${level} ${message}\n`, 'utf8')
    } catch {
      // 日志失败不能影响业务
    }
  }

  private async rotateIfNeeded(): Promise<void> {
    const size = await stat(this.file).then((s) => s.size).catch(() => 0)
    if (size < MAX_BYTES) return
    for (let i = KEEP - 1; i >= 1; i--) {
      await rename(`${this.file}.${i}`, `${this.file}.${i + 1}`).catch(() => {})
    }
    await rename(this.file, `${this.file}.1`).catch(() => {})
  }
}
```

```ts
import type { AnthropicContentBlock, AnthropicMessage, AnthropicRequest, AnthropicResponse, AnthropicTextBlock } from './anthropicTypes'

function systemText(system: AnthropicRequest['system']): string {
  if (!system) return ''
  return typeof system === 'string' ? system : system.map((b) => b.text).join('\n')
}

function resultText(content: string | AnthropicTextBlock[]): string {
  return typeof content === 'string' ? content : content.map((b) => b.text).join('\n')
}

function mapToolChoice(choice: NonNullable<AnthropicRequest['tool_choice']>): unknown {
  if (choice.type === 'auto') return 'auto'
  if (choice.type === 'any') return 'required'
  return { type: 'function', function: { name: choice.name } }
}

export function anthropicToOpenAI(req: AnthropicRequest, upstreamModel: string): Record<string, unknown> {
  const messages: Array<Record<string, unknown>> = []
  const system = systemText(req.system)
  if (system) messages.push({ role: 'system', content: system })

  for (const message of req.messages) {
    if (typeof message.content === 'string') {
      messages.push({ role: message.role, content: message.content })
      continue
    }
    const texts: Array<Record<string, unknown>> = []
    const toolCalls: Array<Record<string, unknown>> = []
    const toolResults: Array<Record<string, unknown>> = []
    for (const block of message.content as AnthropicContentBlock[]) {
      if (block.type === 'text') texts.push({ type: 'text', text: block.text })
      else if (block.type === 'image') {
        const url = block.source.type === 'base64' ? `data:${block.source.media_type};base64,${block.source.data}` : block.source.url
        texts.push({ type: 'image_url', image_url: { url } })
      } else if (block.type === 'tool_use') {
        toolCalls.push({ id: block.id, type: 'function', function: { name: block.name, arguments: JSON.stringify(block.input ?? {}) } })
      } else if (block.type === 'tool_result') {
        toolResults.push({ role: 'tool', tool_call_id: block.tool_use_id, content: resultText(block.content) })
      }
    }
    if (message.role === 'assistant') {
      const out: Record<string, unknown> = { role: 'assistant' }
      if (texts.length) out.content = texts
      if (toolCalls.length) out.tool_calls = toolCalls
      if (!texts.length && !toolCalls.length) out.content = ''
      messages.push(out)
    } else {
      messages.push(...toolResults)
      if (texts.length) messages.push({ role: 'user', content: texts })
    }
  }

  const out: Record<string, unknown> = { model: upstreamModel, messages, max_tokens: req.max_tokens }
  if (req.temperature != null) out.temperature = req.temperature
  if (req.top_p != null) out.top_p = req.top_p
  if (req.stop_sequences?.length) out.stop = req.stop_sequences
  if (req.stream) out.stream = true
  if (req.tools?.length) {
    out.tools = req.tools.map((t) => ({
      type: 'function',
      function: { name: t.name, description: t.description, parameters: t.input_schema }
    }))
  }
  if (req.tool_choice) out.tool_choice = mapToolChoice(req.tool_choice)
  return out
}

function safeJson(text: string): Record<string, unknown> {
  try {
    return JSON.parse(text) as Record<string, unknown>
  } catch {
    return {}
  }
}

function mapFinish(reason: string | null | undefined): AnthropicResponse['stop_reason'] {
  if (reason === 'length') return 'max_tokens'
  if (reason === 'tool_calls') return 'tool_use'
  return 'end_turn'
}

export function openAIToAnthropicResponse(data: unknown, model: string): AnthropicResponse {
  const payload = data as {
    id?: string
    choices?: Array<{ message?: { content?: string; tool_calls?: Array<{ id: string; function: { name: string; arguments: string } }> }; finish_reason?: string }>
    usage?: { prompt_tokens?: number; completion_tokens?: number; prompt_tokens_details?: { cached_tokens?: number } }
  }
  const choice = payload.choices?.[0]
  const content: AnthropicResponse['content'] = []
  if (choice?.message?.content) content.push({ type: 'text', text: choice.message.content })
  for (const call of choice?.message?.tool_calls ?? []) {
    content.push({ type: 'tool_use', id: call.id, name: call.function.name, input: safeJson(call.function.arguments) })
  }
  return {
    id: `msg_${payload.id ?? crypto.randomUUID()}`,
    type: 'message',
    role: 'assistant',
    model,
    content,
    stop_reason: mapFinish(choice?.finish_reason),
    stop_sequence: null,
    usage: {
      input_tokens: payload.usage?.prompt_tokens ?? 0,
      output_tokens: payload.usage?.completion_tokens ?? 0,
      ...(payload.usage?.prompt_tokens_details?.cached_tokens != null
        ? { cache_read_input_tokens: payload.usage.prompt_tokens_details.cached_tokens }
        : {})
    }
  }
}

export function estimateTokens(text: string): number {
  let cjk = 0
  let other = 0
  for (const char of text) {
    if (/[\u4e00-\u9fff\u3040-\u30ff\uac00-\ud7af]/.test(char)) cjk += 1
    else other += 1
  }
  return cjk + Math.ceil(other / 4)
}

export function countTokens(req: AnthropicRequest): number {
  const parts: string[] = [systemText(req.system)]
  for (const message of req.messages) {
    if (typeof message.content === 'string') parts.push(message.content)
    else for (const block of message.content) {
      if (block.type === 'text') parts.push(block.text)
      if (block.type === 'tool_use') parts.push(JSON.stringify(block.input))
      if (block.type === 'tool_result') parts.push(resultText(block.content))
    }
  }
  if (req.tools) parts.push(JSON.stringify(req.tools))
  return estimateTokens(parts.join('\n'))
}
```

```ts
import { parseSSE } from './sse'
import { mapFinish } from './anthropic'

export interface AnthropicStreamOptions {
  model: string
  inputTokens: number
  onUsage?(usage: { promptTokens: number; completionTokens: number }): void
}

export function translateOpenAIStreamToAnthropic(
  source: ReadableStream<Uint8Array>,
  options: AnthropicStreamOptions
): ReadableStream<Uint8Array> {
  const encoder = new TextEncoder()
  return new ReadableStream({
    async start(controller) {
      const send = (event: string, data: unknown) => {
        controller.enqueue(encoder.encode(`event: ${event}\ndata: ${JSON.stringify(data)}\n\n`))
      }
      let nextBlock = 0
      let textBlock: number | null = null
      const toolBlocks = new Map<number, number>()
      let stopReason: string | null = null
      let outputTokens = 0
      let promptTokens = options.inputTokens

      send('message_start', {
        type: 'message_start',
        message: {
          id: `msg_${crypto.randomUUID()}`,
          type: 'message',
          role: 'assistant',
          model: options.model,
          content: [],
          stop_reason: null,
          stop_sequence: null,
          usage: { input_tokens: options.inputTokens, output_tokens: 0 }
        }
      })

      try {
        for await (const event of parseSSE(source)) {
          if (event.data === '[DONE]') break
          let chunk: {
            choices?: Array<{
              delta?: { content?: string; tool_calls?: Array<{ index: number; id?: string; function?: { name?: string; arguments?: string } }> }
              finish_reason?: string | null
            }>
            usage?: { prompt_tokens?: number; completion_tokens?: number }
          }
          try {
            chunk = JSON.parse(event.data)
          } catch {
            continue
          }
          if (chunk.usage?.prompt_tokens != null) promptTokens = chunk.usage.prompt_tokens
          if (chunk.usage?.completion_tokens != null) outputTokens = chunk.usage.completion_tokens
          const choice = chunk.choices?.[0]
          if (!choice) continue
          const delta = choice.delta ?? {}

          if (typeof delta.content === 'string' && delta.content.length > 0) {
            if (textBlock === null) {
              textBlock = nextBlock++
              send('content_block_start', { type: 'content_block_start', index: textBlock, content_block: { type: 'text', text: '' } })
            }
            send('content_block_delta', { type: 'content_block_delta', index: textBlock, delta: { type: 'text_delta', text: delta.content } })
          }

          for (const call of delta.tool_calls ?? []) {
            if (textBlock !== null) {
              send('content_block_stop', { type: 'content_block_stop', index: textBlock })
              textBlock = null
            }
            let index = toolBlocks.get(call.index)
            if (index === undefined) {
              index = nextBlock++
              toolBlocks.set(call.index, index)
              send('content_block_start', {
                type: 'content_block_start',
                index,
                content_block: {
                  type: 'tool_use',
                  id: call.id ?? `toolu_${crypto.randomUUID()}`,
                  name: call.function?.name ?? '',
                  input: {}
                }
              })
            }
            if (call.function?.arguments) {
              send('content_block_delta', { type: 'content_block_delta', index, delta: { type: 'input_json_delta', partial_json: call.function.arguments } })
            }
          }

          if (choice.finish_reason) stopReason = mapFinish(choice.finish_reason)
        }

        if (textBlock !== null) send('content_block_stop', { type: 'content_block_stop', index: textBlock })
        for (const index of [...toolBlocks.values()].sort((a, b) => a - b)) {
          send('content_block_stop', { type: 'content_block_stop', index })
        }
        send('message_delta', {
          type: 'message_delta',
          delta: { stop_reason: stopReason ?? 'end_turn', stop_sequence: null },
          usage: { output_tokens: outputTokens }
        })
        send('message_stop', { type: 'message_stop' })
        options.onUsage?.({ promptTokens, completionTokens: outputTokens })
        controller.close()
      } catch (error) {
        send('error', { type: 'error', error: { type: 'api_error', message: error instanceof Error ? error.message : 'stream failed' } })
        controller.close()
      }
    }
  })
```

```ts
import type { ProviderConfig } from '../types'
import type { RuntimeStore } from '../router/runtime'
import { opencodeGoAdapter } from './opencodeGo'
import { deepseekBalanceAdapter } from './deepseek'
import type { ProbeFetchFn } from './types'

const DEEPSEEK_INTERVAL_MS = 5 * 60 * 1000

export class QuotaScheduler {
  private timer: ReturnType<typeof setInterval> | null = null
  private lastDeepSeek = new Map<string, number>()

  constructor(
    private readonly runtime: RuntimeStore,
    private readonly getProviders: () => ProviderConfig[],
    private readonly getKey: (id: string) => string | null,
    private readonly onUpdate: () => void,
    private readonly now: () => Date = () => new Date(),
    private readonly fetchFn?: ProbeFetchFn
  ) {}

  start(intervalMs: number): void {
    this.stop()
    this.timer = setInterval(() => void this.refreshAll(), intervalMs)
  }

  stop(): void {
    if (this.timer) clearInterval(this.timer)
    this.timer = null
  }

  async refreshAll(force = false): Promise<void> {
    for (const provider of this.getProviders()) {
      if (!provider.enabled || provider.quotaAdapter === 'none') continue
      const key = this.getKey(provider.id)
      if (!key) continue
      const runtime = this.runtime.get(provider.id)
      if (runtime.probeDisabled || runtime.availability === 'auth_error') continue
      if (provider.quotaAdapter === 'deepseek-balance' && !force) {
        const last = this.lastDeepSeek.get(provider.id) ?? 0
        if (this.now().getTime() - last < DEEPSEEK_INTERVAL_MS) continue
      }
      const adapter = provider.quotaAdapter === 'opencode-go' ? opencodeGoAdapter : deepseekBalanceAdapter
      try {
        const snapshot = await adapter.probe(provider, key, { fetchFn: this.fetchFn, now: this.now })
        runtime.probeFailures = 0
        this.runtime.markProbe(provider, snapshot)
        if (provider.quotaAdapter === 'deepseek-balance') this.lastDeepSeek.set(provider.id, this.now().getTime())
      } catch {
        runtime.probeFailures += 1
        if (runtime.probeFailures >= 3) runtime.probeDisabled = true
      }
      this.onUpdate()
    }
  }

  async refreshOne(provider: ProviderConfig, apiKey: string): Promise<void> {
    const adapter = provider.quotaAdapter === 'opencode-go' ? opencodeGoAdapter : deepseekBalanceAdapter
    const snapshot = await adapter.probe(provider, apiKey, { fetchFn: this.fetchFn, now: this.now })
    this.runtime.markProbe(provider, snapshot)
    this.onUpdate()
  }
}
```

实现约束：

- `getState` 返回 `providers` 时附加 `hasKey: this.secrets.hasProviderKey(p.id)`；`runtime` 用 `this.config.providers.map((p) => this.runtime.get(p.id))`；`server` 从 `this.server` 与 `this.paused` 组装（`running: Boolean(this.server)`、`port` 用服务实际端口）；`stats` 用 `await` 不行（同步接口），因此 Hub 维护 `todayStats`，在 `record` 包装里累加；当本地日期变化时先清零再累加。`getState` 直接返回内存值。
- `init` 时读取 `quota-cache.json`（不存在则忽略），把快照写回 `runtime.markProbe`，实现启动即显示额度；同时用 `history.statsFor(today)` 初始化当日统计；若 `secrets.getRecoveryWarning()` 非空，写一条应用日志（不含密钥）。
- `addProvider`：校验 baseUrl（`https://`，或 `http://127.0.0.1` / `http://localhost`）、name、apiKey 非空，否则抛错；用 `createProviderFromPreset` 生成并覆盖字段；保存 Key 与配置后**立即 emit**，再 `await this.discoverFor(provider)`（失败也不阻断保存，记录 logger）；重建目录；启动额度刷新；再次 emit。
- `updateProvider`：只更新传入字段；若传 `apiKey` 非空则更新密钥并 `runtime.reset(id)`（清除旧的鉴权失败状态）；baseUrl/manualModels 变化时发现模型；**任何变更后都执行 `this.catalog = buildCatalog(this.config.providers)`**；保存；emit。
- `deleteProvider`：从配置移除、删除密钥、`runtime.remove`、重建目录、保存、emit。
- `reorderProviders`：检查 ids 与现有集合一致，按 ids 排序保存、**重建目录**、emit。
- `setProviderEnabled`：更新 enabled、保存、**重建目录**、emit。
- `testProvider`：用输入 Key（为空则用已存 Key）拉模型与探针；探针异常返回 `quota: null, error`；不落盘。
- `refreshModels`：对指定或全部供应商发现模型，更新 `cached`/`lastSyncedAt`，失败保留旧值并记录，保存、**重建目录**、emit。
- `refreshQuotas(force)`：委托 `scheduler.refreshAll(force)`，成功快照写入 `quota-cache.json`，emit。
- `refreshProviderQuota(id)`：对有探针适配器且有 Key 的供应商调用 `scheduler.refreshOne`，快照写入 `quota-cache.json`，emit；**必须 try/catch**，探针抛错时记录日志并返回，不得把异常抛给 IPC。
- `refreshStaleModels`：对 `lastSyncedAt` 为 null 或早于 24h 的供应商调用发现。
- `resetProvider`：`runtime.reset(id)`，emit。
- `updateSettings`：端口变化时先尝试 `startServer` 新端口；成功则关闭旧服务并保存；失败则抛错并保留旧服务。其余字段直接保存。autostart 由 Electron 层处理。

```ts
import { deflateSync } from 'node:zlib'

const CRC_TABLE = new Uint32Array(256).map((_, n) => {
  let c = n
  for (let k = 0; k < 8; k++) c = c & 1 ? 0xedb88320 ^ (c >>> 1) : c >>> 1
  return c
})

function crc32(buf: Buffer): number {
  let c = 0xffffffff
  for (const byte of buf) c = CRC_TABLE[(c ^ byte) & 0xff] ^ (c >>> 8)
  return (c ^ 0xffffffff) >>> 0
}

function chunk(type: string, data: Buffer): Buffer {
  const length = Buffer.alloc(4)
  length.writeUInt32BE(data.length)
  const typeAndData = Buffer.concat([Buffer.from(type), data])
  const crc = Buffer.alloc(4)
  crc.writeUInt32BE(crc32(typeAndData))
  return Buffer.concat([length, typeAndData, crc])
}

export function circlePng(size: number, rgb: [number, number, number]): Buffer {
  const header = Buffer.alloc(13)
  header.writeUInt32BE(size, 0)
  header.writeUInt32BE(size, 4)
  header[8] = 8
  header[9] = 6
  const raw = Buffer.alloc(size * (size * 4 + 1))
  for (let y = 0; y < size; y++) {
    const row = y * (size * 4 + 1)
    for (let x = 0; x < size; x++) {
      const cx = x - size / 2 + 0.5
      const cy = y - size / 2 + 0.5
      const inside = cx * cx + cy * cy <= (size / 2 - 1) ** 2
      const offset = row + 1 + x * 4
      raw[offset] = rgb[0]
      raw[offset + 1] = rgb[1]
      raw[offset + 2] = rgb[2]
      raw[offset + 3] = inside ? 255 : 0
    }
  }
  return Buffer.concat([
    Buffer.from([137, 80, 78, 71, 13, 10, 26, 10]),
    chunk('IHDR', header),
    chunk('IDAT', deflateSync(raw)),
    chunk('IEND', Buffer.alloc(0))
  ])
}
```

```ts
import { Hub } from '../core/hub'
import { safeStorageCipher } from './safeStorageCipher'

let tray: Tray | null = null
let hub: Hub | null = null
let quitting = false

const dataDir = process.env.TOKENHUB_DATA_DIR
if (dataDir) app.setPath('userData', dataDir)
if (!app.requestSingleInstanceLock()) app.quit()

function trayColor(state: AppState): [number, number, number] {
  const exhausted = state.runtime.some((r) => r.availability === 'exhausted' || r.availability === 'rate_limited')
  const noneRoutable = state.providers.every((p) => {
    if (!p.enabled) return true
    const rt = state.runtime.find((r) => r.providerId === p.id)
    if (!rt) return false
    if (rt.availability === 'auth_error' || rt.availability === 'exhausted') return true
    return Boolean(rt.cooldownUntil && Date.parse(rt.cooldownUntil) > Date.now())
  })
  if (noneRoutable) return [239, 68, 68]
  return exhausted ? [234, 179, 8] : [34, 197, 94]
}

function updateTray(state: AppState): void {
  if (!tray) return
  tray.setImage(nativeImage.createFromBuffer(circlePng(32, trayColor(state))))
  tray.setToolTip(`TokenHUB · 127.0.0.1:${state.server.port}`)
  tray.setContextMenu(Menu.buildFromTemplate([
    { label: '打开主界面', click: () => { win?.show(); win?.focus() } },
    {
      label: state.server.paused ? '恢复路由' : '暂停路由',
      click: () => {
        hub?.setPaused(!state.server.paused)
        if (hub) updateTray(hub.getState())
      }
    },
    { label: '复制 OpenAI 地址', click: () => clipboard.writeText(`http://127.0.0.1:${state.server.port}/v1`) },
    { label: '复制 Anthropic 地址', click: () => clipboard.writeText(`http://127.0.0.1:${state.server.port}`) },
    { type: 'separator' },
    { label: '退出', click: () => { quitting = true; app.quit() } }
  ]))
}

async function boot(): Promise<void> {
  hub = new Hub({
    dataDir: app.getPath('userData'),
    cipher: safeStorageCipher,
    portOverride: process.env.TOKENHUB_PORT ? Number(process.env.TOKENHUB_PORT) : undefined
  })
  await hub.init()
  registerIpc(hub, (state) => {
    win?.webContents.send('tokenhub:state', state)
    updateTray(state)
  })
  const state = hub.getState()
  win = new BrowserWindow({
    width: 1180,
    height: 760,
    minWidth: 960,
    minHeight: 640,
    show: !state.settings.startMinimized,
    backgroundColor: '#f7f7f5',
    webPreferences: {
      preload: join(__dirname, '../preload/index.js'),
      contextIsolation: true,
      nodeIntegration: false,
      sandbox: true
    }
  })
  win.on('close', (event) => {
    if (!quitting && hub?.getState().settings.closeToTray) {
      event.preventDefault()
      win?.hide()
    }
  })
  if (process.env.ELECTRON_RENDERER_URL) await win.loadURL(process.env.ELECTRON_RENDERER_URL)
  else await win.loadFile(join(__dirname, '../renderer/index.html'))
  tray = new Tray(nativeImage.createFromBuffer(circlePng(32, [34, 197, 94])))
  updateTray(state)
  app.setLoginItemSettings({ openAtLogin: state.settings.autoStart })
}

app.whenReady().then(boot).catch((error) => {
  dialog.showErrorBox('TokenHUB 启动失败', error instanceof Error ? error.message : String(error))
  app.quit()
})

app.on('second-instance', () => {
  win?.show()
  win?.focus()
})
app.on('window-all-closed', () => {
  if (!hub?.getState().settings.closeToTray) app.quit()
})
app.on('before-quit', async () => {
  quitting = true
  await hub?.shutdown()
})
```
